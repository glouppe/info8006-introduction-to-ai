\begin{recap}
    \topic{Bayesian networks}
    With a partially observable environment, an agent reasons about the state $s$ from its percepts $e$ with the joint distribution $P(s, e) = P(s) P(e \mid s)$. When $s$ and $e$ are assignments of many random variables $X_1, \ldots, X_n$, a \term{Bayesian network} represents this distribution compactly. It is a directed acyclic graph with one node per variable, each node $X_i$ annotated with a conditional distribution $\mathbf{P}(X_i \mid \text{parents}(X_i))$, given as a \term{conditional probability table} (CPT) for discrete variables. The topology asserts the \term{local Markov property}, $X_i \perp \text{non-descendants}(X_i) \mid \text{parents}(X_i)$, hence the factorization
    \[
        P(x_1, \ldots, x_n) = \prod_{i=1}^n P(x_i \mid \text{parents}(X_i)).
    \]
    Naive Bayes is the network in which a cause $C$ is the only parent of each effect $E_i$.

    \topic{d-separation}
    $X_i \perp X_j \mid Z_1, \ldots, Z_m$ is guaranteed if every undirected path between $X_i$ and $X_j$ is \term{inactive}; if one path is \term{active}, independence is not guaranteed. A path is active iff each of its triples is active:
    \begin{center}
        \begin{tabular}{lll}
            \toprule
            Triple & & Active iff \\
            \midrule
            Cascade & $A \to B \to C$ or $A \leftarrow B \leftarrow C$ & $B$ unobserved \\
            Common parent & $A \leftarrow B \to C$ & $B$ unobserved \\
            v-structure & $A \to B \leftarrow C$ & $B$ or a descendant of $B$ observed \\
            \bottomrule
        \end{tabular}
    \end{center}
    Given its \term{Markov blanket}, \ie{} its parents, its children and the other parents of its children, a variable is independent of all the others.

    \topic{Inference}
    Given \term{evidence} variables $\mathbf{E}$ observed as $\mathbf{e}$, a \term{query} variable $Q$ and \term{hidden} variables $\mathbf{H}$, inference computes
    \[
        \mathbf{P}(Q \mid \mathbf{e}) = \frac{1}{Z} \sum_{\mathbf{h}} \mathbf{P}(Q, \mathbf{h}, \mathbf{e}), \qquad Z = \sum_q \sum_{\mathbf{h}} P(q, \mathbf{h}, \mathbf{e}),
    \]
    with the joint entries written as products of CPT entries. \term{Inference by enumeration} evaluates the sums depth-first, in $O(n)$ space and $O(d^n)$ time for $n$ variables with $d$ values. \term{Variable elimination} evaluates them right to left and stores the intermediate \term{factors}, multi-dimensional arrays initialized with the CPTs given the evidence. It alternates the \term{join} of factors, their pointwise product over the union of their variables, and the \term{elimination} of a hidden variable, summed out of the join of the factors that mention it. Its cost is that of the largest factor, which depends on the elimination ordering; finding the best ordering is NP-hard. A hidden variable $H$ is \term{irrelevant}, and can be removed, unless $H \in \text{ancestors}(\{Q\} \cup \mathbf{E})$.

    \topic{Parameter learning}
    The CPTs can be chosen in a \term{parameterized} family $\mathbf{P}(X \mid \theta)$, whose parameters are estimated from $N$ i.i.d.\ observations $\mathbf{d} = \{x_1, \ldots, x_N\}$. The \term{maximum likelihood estimate} (MLE) maximizes the \term{likelihood} $P(\mathbf{d} \mid \theta) = \prod_{j=1}^N P(x_j \mid \theta)$, or the \term{log-likelihood} $L(\theta) = \log P(\mathbf{d} \mid \theta)$, often by setting $\frac{\partial L}{\partial \theta} = 0$:
    \[
        \theta^*_{\text{MLE}} = \arg \max_\theta P(\mathbf{d} \mid \theta).
    \]
    \term{Bayesian parameter learning} treats $\theta$ as a random variable with a \term{prior} $\mathbf{P}(\theta)$, updated into the \term{posterior} $\mathbf{P}(\theta \mid \mathbf{d}) \propto \mathbf{P}(\mathbf{d} \mid \theta) \mathbf{P}(\theta)$. A new observation $x$ is then predicted by the \term{posterior predictive} $p(x \mid \mathbf{d}) = \int p(x \mid \theta) p(\theta \mid \mathbf{d}) \, d\theta$, and by the \term{prior predictive} $p(x) = \int p(x \mid \theta) p(\theta) \, d\theta$ before any data. When the posterior is intractable, the \term{maximum a posteriori} (MAP) estimate replaces it by its mode,
    \[
        \theta^*_{\text{MAP}} = \arg \max_\theta P(\theta \mid \mathbf{d}) = \arg \max_\theta P(\mathbf{d} \mid \theta) P(\theta).
    \]

    \topic{Cheat sheet for Gaussian distributions}
    The Gaussian density of mean $\mu$ and variance $\sigma^2$ is
    \[
        \mathcal{N}(x \mid \mu, \sigma^2) = \frac{1}{\sqrt{2 \pi \sigma^2}} \exp\left(-\frac{(x - \mu)^2}{2 \sigma^2}\right).
    \]
    If $\mathbf{x}$ and $\mathbf{y}$ have the joint Gaussian distribution
    \[
        p\begin{pmatrix}
            \mathbf{x} \\
            \mathbf{y}
        \end{pmatrix}
        = \mathcal{N}\left(
        \begin{pmatrix}
            \mathbf{x} \\
            \mathbf{y}
        \end{pmatrix}
        \,\middle|\,
        \begin{pmatrix}
            \mathbf{a} \\
            \mathbf{b}
        \end{pmatrix},
        \begin{pmatrix}
            \mathbf{A} & \mathbf{C} \\
            \mathbf{C}^T & \mathbf{B}
        \end{pmatrix}
        \right),
    \]
    then the marginal and conditional distributions of $\mathbf{x}$ and $\mathbf{y}$ are given by
    \begin{align*}
        p(\mathbf{x}) & = \mathcal{N}(\mathbf{x} \mid \mathbf{a}, \mathbf{A}) \\
        p(\mathbf{y}) & = \mathcal{N}(\mathbf{y} \mid \mathbf{b}, \mathbf{B}) \\
        p(\mathbf{x} \mid \mathbf{y}) & = \mathcal{N}(\mathbf{x} \mid \mathbf{a} + \mathbf{C} \mathbf{B}^{-1} (\mathbf{y} - \mathbf{b}), \mathbf{A} - \mathbf{C} \mathbf{B}^{-1} \mathbf{C}^T) \\
        p(\mathbf{y} \mid \mathbf{x}) & = \mathcal{N}(\mathbf{y} \mid \mathbf{b} + \mathbf{C}^T \mathbf{A}^{-1} (\mathbf{x} - \mathbf{a}), \mathbf{B} - \mathbf{C}^T \mathbf{A}^{-1} \mathbf{C}).
    \end{align*}
    Conversely, if $\mathbf{x}$ and $\mathbf{y}$ have the Gaussian distributions
    \begin{align*}
        p(\mathbf{x}) & = \mathcal{N}(\mathbf{x} \mid \mathbf{m}, \mathbf{P}) \\
        p(\mathbf{y} \mid \mathbf{x}) & = \mathcal{N}(\mathbf{y} \mid \mathbf{H} \mathbf{x} + \mathbf{u}, \mathbf{R}),
    \end{align*}
    then their joint distribution is Gaussian with
    \[
        p\begin{pmatrix}
            \mathbf{x} \\
            \mathbf{y}
        \end{pmatrix}
        = \mathcal{N}\left(
        \begin{pmatrix}
            \mathbf{x} \\
            \mathbf{y}
        \end{pmatrix}
        \,\middle|\,
        \begin{pmatrix}
            \mathbf{m} \\
            \mathbf{H} \mathbf{m} + \mathbf{u}
        \end{pmatrix},
        \begin{pmatrix}
            \mathbf{P} & \mathbf{P} \mathbf{H}^T \\
            \mathbf{H} \mathbf{P} & \mathbf{H} \mathbf{P} \mathbf{H}^T + \mathbf{R}
        \end{pmatrix}
        \right).
    \]
\end{recap}

\titledquestion{Nuclear power plant \aima{13}}

A monitoring program supervises the core of a nuclear power station. An alarm sounds when a temperature gauge, which measures the temperature of the core, exceeds a given threshold, and the program reasons about the actual temperature from the gauge and the alarm, either of which may be faulty. Consider the Boolean variables $A$ (alarm sounds), $F_A$ (alarm is faulty), and $F_G$ (gauge is faulty) and the multi-valued nodes $G$ (gauge reading) and $T$ (actual core temperature).

\begin{parts}
    \part Draw a Bayesian network for this domain, given that the gauge is more likely to fail when the core temperature gets too high.

    \begin{solution}
        \begin{center}
            \begin{tikzpicture}[bn diagram]
                \node[random variable] (T) at (0, 0) {$T$};
                \node[random variable] (FG) at (1.5, 1.3) {$F_G$};
                \node[random variable] (G) at (3, 0) {$G$};
                \node[random variable] (FA) at (4.5, 1.3) {$F_A$};
                \node[random variable] (A) at (4.5, 0) {$A$};
                \draw[dependency] (T) -- (FG);
                \draw[dependency] (T) -- (G);
                \draw[dependency] (FG) -- (G);
                \draw[dependency] (G) -- (A);
                \draw[dependency] (FA) -- (A);
            \end{tikzpicture}
        \end{center}
    \end{solution}

    \part From your network topology, can you conclude $T \perp F_A \mid F_G$ and $T \perp F_A \mid F_G, A$?

    \begin{solution}
        The paths from $T$ to $F_A$ are ($T, F_G, G, A, F_A$) and ($T, G, A, F_A$).

        Knowing $F_G$, the triple ($G, A, F_A$) is inactive (v-structure, $A$ unobserved). Since it is present in both paths, there is no active path, so $T \perp F_A \mid F_G$ is true.

        For the second relation, the triple ($G, A, F_A$) becomes active as we observe $A$. The only way we can ensure conditional independence is by ensuring that at least one remaining triple is inactive in each path. For the first path, we have ($T, F_G, G$) which is inactive (cascade) which makes the path inactive. However, for the second path, ($T, G, A$) is active (cascade) which makes the path active. We have at least one active path from $T$ to $F_A$ which prevents us from concluding the conditional independence in this case.
    \end{solution}

    \part Suppose there are just two possible actual and measured temperatures: low ($l$) and high ($h$). The probability that the gauge gives the correct temperature is $x$ when it is working, but $y$ when it is faulty. Give the conditional probability table associated with $G$.

    \begin{solution}
        The CPT for $G$ is shown below. Note the semantics of $F_G$, which is true when the gauge is faulty, \ie{} not working properly.
        \begin{center}
            \begin{tabular}{ccc}
                \toprule
                $T$ & $F_G$ & $P(G = l \mid T, F_G)$ \\
                \midrule
                $l$ & 0 & $x$ \\
                $l$ & 1 & $y$ \\
                $h$ & 0 & $1 - x$ \\
                $h$ & 1 & $1 - y$ \\
                \bottomrule
            \end{tabular}
        \end{center}
    \end{solution}

    \part Suppose the alarm is always triggered by high measured temperatures, unless it is faulty, in which case it never sounds. Give the conditional probability table associated with $A$.

    \begin{solution}
        \begin{center}
            \begin{tabular}{ccc}
                \toprule
                $G$ & $F_A$ & $P(A = 1 \mid G, F_A)$ \\
                \midrule
                $l$ & 0 & 0 \\
                $l$ & 1 & 0 \\
                $h$ & 0 & 1 \\
                $h$ & 1 & 0 \\
                \bottomrule
            \end{tabular}
        \end{center}
    \end{solution}

    \part Suppose the gauge is not faulty and the alarm is triggered. Calculate an expression for the probability that the temperature of the core is too high, in terms of the various conditional probabilities in the network.

    \begin{solution}
        The probability of interest here is $P(T = h \mid A = 1, F_G = 0)$. Because the alarm's behavior is deterministic, we can deduce that $G$ indicates high ($h$). Hence, $P(T = h \mid A = 1, F_G = 0) = P(T = h \mid A = 1, G = h, F_G = 0)$. We also see that $A \perp T \mid G$. Therefore, we only need to calculate
        \begin{align*}
            P(T = h \mid G = h, F_G = 0) & = \frac{P(T = h, G = h, F_G = 0)}{P(G = h, F_G = 0)} \\
            & = \frac{P(G = h \mid T = h, F_G = 0) P(F_G = 0 \mid T = h) P(T = h)}{\sum_t P(G = h \mid t, F_G = 0) P(F_G = 0 \mid t) P(t)} ,
        \end{align*}
        which we cannot develop more as we don't know $P(T)$ and $P(F_G \mid T)$.
    \end{solution}
\end{parts}

\titledquestion{Energy of an inference server}

A scheduling program manages a data center in which a GPU server answers the requests sent to a language model. To plan the power supply, the scheduler models the energy the server consumes as a function of its load, with the following network

\begin{center}
    \begin{tikzpicture}[bn diagram]
        \node[observed variable] (S) at (0, 0) {$S$};
        \node[observed variable] (H) at (2.5, 0) {$H$};
        \draw[dependency] (S) -- (H);
    \end{tikzpicture}
\end{center}

where $S$ is a random variable denoting the number of requests the server receives in an hour, and $H$ a random variable denoting the energy, in kilowatt-hours, that it consumes during that hour. The scheduler chooses the following relations for each variable
\begin{equation}
    \label{P_S}
    P(S = s) = \texttt{Poisson}(S = s; \lambda) = \dfrac{\lambda^s e^{-\lambda}}{s!}
\end{equation}
\begin{equation}
    \label{P_H_g_S}
    H = \omega S + H_0 + \mathcal{N}(0, \var)
\end{equation}
where $H_0$ is the energy the server consumes when idle, which is known, and $\omega$ is the energy consumed per request.

\begin{parts}
    \part Identify the parameters of the Bayesian network and give the joint log-likelihood of one pair $(s_i, h_i)$ given the model.

    \begin{solution}
        Given the topology of the network, we have to identify $P(S = s)$ and $P(H = h \mid S = s)$. The former is given by~(\ref{P_S}) whereas the latter can be deduced from~(\ref{P_H_g_S}). We have
        \begin{equation*}
            P(S = s) = \texttt{Poisson}(S = s; \lambda)
        \end{equation*}
        and
        \begin{equation*}
            p(h \mid s) = \mathcal{N}(h \mid \omega s + H_0, \var).
        \end{equation*}

        The parameters of those distributions are $\theta = [\lambda, \omega, \sigma]$.

        The likelihood of a pair $(s_i, h_i)$ can be derived from the network. We have
        \begin{align*}
            P((s_i, h_i) \mid \theta) & = P(s_i) P(h_i \mid s_i) \\
            & = \dfrac{\lambda^{s_i} e^{-\lambda}}{s_i!} \dfrac{1}{\sqrt{2\pi}\sigma} e^{-\frac{(h_i - \omega s_i - H_0)^2}{2\var}}
        \end{align*}
        from which we derive the log-likelihood
        \begin{equation}
            L((s_i, h_i); \theta) = s_i \log{\lambda} - \lambda - \log{s_i!} - \log{\sqrt{2\pi}\sigma} - \dfrac{(h_i - \omega s_i - H_0)^2}{2\var}.
        \end{equation}
    \end{solution}

    \part The scheduler collects a set $\mathbf{d} = \{(s_i, h_i)\}_{i=1}^N$ of independent observations. Derive the maximum likelihood estimate of the parameters.

    \begin{solution}
        We first have to define the joint likelihood of those observations. Since they are assumed i.i.d., we have
        \[
            P(\mathbf{d} \mid \theta) = \prod_{i=1}^N P((s_i, h_i) \mid \theta)
        \]
        Hence, deriving the log-likelihood
        \begin{align*}
            L(\mathbf{d}; \theta) & = \sum_i L((s_i, h_i); \theta) \\
            & = -N(\lambda + \log{\sqrt{2\pi}\sigma}) + \sum_i s_i \log{\lambda} - \log{s_i!} - \dfrac{(h_i - \omega s_i - H_0)^2}{2\var}
        \end{align*}
        The maximum likelihood parameters can be found by cancelling the derivative of the log-likelihood w.r.t. each parameter.

        For $\lambda$ we have
        \begin{align*}
            \dfrac{\partial L}{\partial \lambda} & = -N + \sum_i \dfrac{s_i}{\lambda} \\
            & = 0
        \end{align*}
        which gives $\boxed{\lambda_{\text{MLE}} = \dfrac{1}{N} \sum_i s_i}$.

        For $\sigma$ we have
        \begin{align*}
            \dfrac{\partial L}{\partial \sigma} & = -\dfrac{1}{\sigma} \bigg(N - \sum_i \dfrac{(h_i - \omega s_i - H_0)^2}{\var}\bigg) \\
            & \propto N - \dfrac{1}{\var} \sum_i (h_i - \omega s_i - H_0)^2 \\
            & = 0
        \end{align*}
        which gives $\boxed{\var_{\text{MLE}} = \dfrac{1}{N} \sum_i (h_i - \omega s_i - H_0)^2}$.

        For $\omega$ we have
        \begin{align*}
            \dfrac{\partial L}{\partial \omega} & = \sum_i \dfrac{s_i (h_i - \omega s_i - H_0)}{\var} \\
            & \propto \bigg(\sum_i s_i (h_i - H_0)\bigg) - \omega \sum_i s_i^2 \\
            & = 0
        \end{align*}
        which gives $\boxed{\omega_{\text{MLE}} = \dfrac{\sum_i s_i (h_i - H_0)}{\sum_i s_i^2}}$.
    \end{solution}

    \part The scheduler has just started to collect observations, and maximum likelihood can overfit when they are few. It therefore treats the parameter $\lambda$, about which it has no strong knowledge, as uncertain. Using a prior $\lambda \sim \texttt{Gamma}(\alpha, \beta)$, what would be the posterior distribution $p(\lambda \mid \mathbf{s})$, with $\mathbf{s} = \{s_i\}_{i=1}^N$?

    \textit{Hint:} $\texttt{Gamma}(x \mid \alpha, \beta) = \frac{1}{Z} x^{\alpha - 1} e^{-\beta x}$, with $Z$ a normalizing constant.

    \begin{solution}
        Using Bayes' rule
        \begin{align*}
            p(\lambda \mid \mathbf{s}) & \propto P(\mathbf{s} \mid \lambda) p(\lambda) \\
            & \propto \lambda^{\alpha - 1} e^{-\beta \lambda} \prod_i \dfrac{\lambda^{s_i} e^{-\lambda}}{s_i!} \\
            & = \lambda^{\alpha - 1} e^{-\beta \lambda} \bigg(\dfrac{\lambda^{\sum_i s_i} e^{-N\lambda}}{\prod_i s_i!}\bigg) \\
            & \propto \lambda^{(\alpha + \sum_i s_i - 1)} e^{-(\beta + N) \lambda}
        \end{align*}
        which corresponds to $\texttt{Gamma}(\alpha + \sum_i s_i,\ \beta + N)$, the posterior density. Note that $\propto$ stands for the normalizing constant of the posterior density. Everything that does not depend explicitly on the parameter $\lambda$ can be absorbed in the constant if it multiplies the whole.
    \end{solution}

    \part Knowing that the mean, the variance and the mode of $\texttt{Gamma}(\alpha, \beta)$ are respectively $\dfrac{\alpha}{\beta}$, $\dfrac{\alpha}{\beta^2}$ and $\dfrac{\alpha - 1}{\beta}$, interpret how they evolve w.r.t. $N$ between the prior and the posterior.

    \begin{solution}
        We have
        \[
            \E(\lambda): \dfrac{\alpha}{\beta} \Rightarrow \dfrac{\alpha + \sum_i s_i}{\beta + N},
        \]
        \[
            \V(\lambda): \dfrac{\alpha}{\beta^2} \Rightarrow \dfrac{\alpha + \sum_i s_i}{(\beta + N)^2}
        \]
        and
        \[
            \underset{\lambda}{\arg \max}\, p(\lambda \mid \mathbf{s}): \dfrac{\alpha - 1}{\beta} \Rightarrow \dfrac{\alpha + \sum_i s_i - 1}{\beta + N}.
        \]
        When $N = 0$, the prior and the posterior are similar, which is expected. When $N \rightarrow \infty$, knowing that $s \in \mathbb{N}$ by definition (a number of requests), we can assume that, for reasonable values of $\alpha, \beta$, we have $\sum_i s_i \gg \alpha$ and $N \gg \beta$. This implies that the mean and the mode converge to the same value corresponding to $\lambda_{\text{MLE}}$. The variance decreases towards 0, which suggests that $\lambda_{\text{MLE}}$ is the best and unique estimate of $\lambda$ for an infinite amount of observations. This illustrates the trade-off between likelihood and prior in a Bayesian parameter learning setting. As more data are observed, the uncertainty about a parameter decreases smoothly from the prior to the likelihood.
    \end{solution}
\end{parts}

\titledquestion{Evolutionary model merging \aima{13}}

Evolutionary model merging builds a new language model $M$ from two parent models $A$ and $B$: each component of $M$ is copied from one of the parents. Consider one component, and let $G_x$ be the language it was mostly trained on in model $x$, with possible values $e$ (English) or $f$ (French). A common \emph{hypothesis} is that the default language of a model follows from this component by a simple mechanism: the language $H_x$ in which model $x$ answers by default is the same as $G_x$ with some probability $s$.

Furthermore, $M$ copies the component from either parent with equal probability, and a random perturbation flips it with a small nonzero probability $m$.

\begin{parts}
    \part Which of the following networks claim that $\mathbf{P}(G_M \mid G_A, G_B) = \mathbf{P}(G_M)$? Shaded variables are observed.

    \begin{center}
        \begin{tikzpicture}[bn diagram]
            \node[random variable] (GA1) at (0, 0) {$G_A$};

\titledquestion{Gaussian parameter learning with known $\var$}

You have in your possession a high-tech laser that emits light of intensity $\mu_l$ with great precision. You want to use it to test a new light-intensity measurement device. You assume that your measures $y_i$ follow a Gaussian distribution $\N(\mu_y, \var)$ with $\mu_y \sim \N(\mu_l, \sigma_l^2)$ corresponding to the model of your laser given in the manual. You assume that $\var$ is fixed and that you can determine it.

You have access to $\mathcal{Y}_c$ a calibration set of $C$ observations and you want to make $N$ new measures with your device. You want to study how you can reduce your uncertainty about the measures using $\mathcal{Y}_c$.

\begin{parts}
    \part Compute the prior and posterior predictive distributions $P(y)$ and $P(y \mid \mathcal{Y}_c)$ if you have access to one observation, \ie{} $\mathcal{Y}_c = \{y_o\}$.

    \begin{solution}
        The following exercise uses the Gaussian model identities (see the cheat sheet of the recap).

        We first identify the components we have:
        \begin{itemize}
            \item $p(y \mid \mu_y) = \N(y \mid \mu_y, \var)$
            \item $p(\mu_y) = \N(\mu_y \mid \mu_l, \sigma_l^2)$
        \end{itemize}

        \textbf{Prior predictive}
        \[
            P(y) = \int P(y \mid \mu_y) P(\mu_y) d\mu_y.
        \]
        From Gaussian identities, you can identify different parameters as follows
        \begin{align*}
            \mathbf{x} & = \mu_y \\
            \mathbf{m} & = \mu_l \\
            \mathbf{P} & = \sigma_l^2 \\
            \mathbf{y} & = y \\
            \mathbf{H} & = 1 \\
            \mathbf{u} & = 0 \\
            \mathbf{R} & = \var
        \end{align*}
        which corresponds to the joint
        \[
            P(\mu_y, y) = \N
            \begin{pmatrix}
                \begin{pmatrix}
                    \mu_l \\
                    \mu_l \\
                \end{pmatrix},
                \begin{pmatrix}
                    \sigma_l^2 & \sigma_l^2 \\
                    \sigma_l^2 & \sigma_l^2 + \var \\
                \end{pmatrix}
            \end{pmatrix}
        \]
        from which we extract $p(y) = \N(y \mid \mu_l, \var + \sigma_l^2)$.

        \textbf{Posterior predictive}
        \[
            P(y \mid \mathcal{Y}_c) = \int P(y \mid \mu_y) P(\mu_y \mid \mathcal{Y}_c) d\mu_y.
        \]

        We first have to identify the posterior $P(\mu_y \mid \mathcal{Y}_c)$. Once again, from Gaussian identities, knowing $P(\mathcal{Y}_c \mid \mu_y)$ and $P(\mu_y)$ you can identify different parameters as follows
        \begin{align*}
            \mathbf{x} & = \mu_y \\
            \mathbf{m} & = \mu_l \\
            \mathbf{P} & = \sigma_l^2 \\
            \mathbf{y} & = y_o \\
            \mathbf{H} & = 1 \\
            \mathbf{u} & = 0 \\
            \mathbf{R} & = \var
        \end{align*}
        which corresponds to the joint
        \[
            P(\mu_y, y_o) = \N
            \begin{pmatrix}
                \begin{pmatrix}
                    \mu_l \\
                    \mu_l \\
                \end{pmatrix},
                \begin{pmatrix}
                    \sigma_l^2 & \sigma_l^2 \\
                    \sigma_l^2 & \sigma_l^2 + \var \\
                \end{pmatrix}
            \end{pmatrix}
        \]
        from which we extract
        \begin{align*}
            p(\mu_y \mid y_o) & = \N(\mu_y \mid \mu_{\text{post}}, \sigma_{\text{post}}^2) \\
            & = \N\bigg(\mu_y \,\bigg|\, \mu_l + \dfrac{\sigma_l^2}{\var + \sigma_l^2} (y_o - \mu_l), \sigma_l^2 - \dfrac{\sigma_l^4}{\var + \sigma_l^2}\bigg).
        \end{align*}
        After simplification, we identify the posterior parameters as
        \[
            \mu_{\text{post}} = \dfrac{\var}{\var + \sigma_l^2} \mu_l + \dfrac{\sigma_l^2}{\var + \sigma_l^2} y_o
        \]
        and
        \[
            \sigma_{\text{post}}^2 = \dfrac{\var}{\sigma_l^2 + \var} \sigma_l^2.
        \]
        Following the same development as for the prior predictive, we have
        \[
            p(y \mid \mathcal{Y}_c) = \N(y \mid \mu_{\text{post}}, \var + \sigma_{\text{post}}^2).
        \]
    \end{solution}

    \part Looking at your predictive uncertainty, what is the benefit of incorporating $\mathcal{Y}_c$?

    \begin{solution}
        As a reminder, we found
        \[
            p(y) = \N(y \mid \mu_l, \var + \sigma_l^2).
        \]
        \[
            p(y \mid \mathcal{Y}_c) = \N(y \mid \mu_{\text{post}}, \var + \sigma_{\text{post}}^2).
        \]
        Comparing only the variances, we have
        \begin{align*}
            \var + \sigma_l^2 & \overset{?}{\lessgtr} \var + \sigma_{\text{post}}^2 \\
            \sigma_l^2 & \overset{?}{\lessgtr} \sigma_{\text{post}}^2 \\
            1 & \overset{?}{\lessgtr} \dfrac{\var}{\sigma_l^2 + \var} \\
            \sigma_l^2 & \geq 0
        \end{align*}
        and we deduce that the posterior predictive has a smaller or equal variance to the prior one. We conclude that incorporating the calibration set can reduce uncertainty about the measures.

        If we computed the variance for a calibration set of $C$ observations, we would have found a posterior variance
        \[
            \sigma_{\text{post}}^2 = \dfrac{\var}{C\sigma_l^2 + \var} \sigma_l^2.
        \]
        Since $\var$ is fixed, we see that including more observations reduces even more the posterior predictive variance. In the limit case $C \to \infty$, the posterior predictive variance converges to $\var$ which corresponds to the uncertainty of the instrument (you can't reach a lower uncertainty, it is called the aleatoric uncertainty).
    \end{solution}
\end{parts}

\titledquestion{Independence and variable elimination}

Considering the Bayesian network below, which of the following statements are enforced by the network structure?

\begin{center}
    \begin{tikzpicture}[bn diagram]
        \node[random variable] (A) at (0, 0) {$A$};
        \node[random variable] (F) at (-1.5, -1.5) {$F$};
        \node[random variable] (B) at (0, -1.5) {$B$};
        \node[random variable] (C) at (1.5, -1.5) {$C$};
        \node[random variable] (D) at (0.75, -3) {$D$};
        \node[random variable] (E) at (2.25, -3) {$E$};
        \draw[dependency] (A) -- (B);
        \draw[dependency] (A) -- (C);
        \draw[dependency] (B) -- (D);
        \draw[dependency] (C) -- (D);
        \draw[dependency] (C) -- (E);
        \draw[dependency] (A) -- (F);
        \draw[dependency] (B) -- (F);
    \end{tikzpicture}
\end{center}

\begin{solution}
    We apply the d-separation algorithm. To show that two variables $X$ and $Y$ could be dependent, it is sufficient to find a single \emph{active} (undirected) path from $X$ to $Y$. A path is active if all of its consecutive triples are active.
\end{solution}

\begin{parts}
    \part $\mathbf{P}(B, C) = \mathbf{P}(B) \mathbf{P}(C)$

    \begin{solution}
        This is true iff (if and only if) $B \perp C$ ($B$ is independent from $C$). The three paths that link $B$ and $C$ are $(B, A, C)$, $(B, D, C)$ and $(B, F, A, C)$. $(B, A, C)$ is active because $A$ is unobserved (common parent). We cannot guarantee that $B \perp C$.
    \end{solution}

    \part $\mathbf{P}(B, C \mid A) = \mathbf{P}(B \mid A) \mathbf{P}(C \mid A)$

    \begin{solution}
        This is true iff $B \perp C \mid A$. This time, $(B, A, C)$ is not active because $A$ is known. $(B, D, C)$ is not active either because $D$ is unobserved (v-structure). In $(B, F, A, C)$, the triple $(B, F, A)$ is not active (v-structure). $(F, A, C)$ is not active either. We can guarantee that $B \perp C \mid A$.
    \end{solution}

    \part $\mathbf{P}(F, E) = \mathbf{P}(F) \mathbf{P}(E)$

    \begin{solution}
        This is true iff $F \perp E$. The four paths that link $F$ and $E$ are $(F, A, C, E)$, $(F, A, B, D, C, E)$, $(F, B, A, C, E)$ and $(F, B, D, C, E)$. In $(F, A, C, E)$, the triples $(F, A, C)$ (common parent) and $(A, C, E)$ (cascade) are both active. We cannot guarantee that $F \perp E$.
    \end{solution}

    \part $\mathbf{P}(F \mid A, D, E) = \mathbf{P}(F \mid A, D)$

    \begin{solution}
        This is true iff $F \perp E \mid A, D$. This time, the triples $(F, A, C)$, $(F, A, B)$ and $(B, A, C)$ are inactive (common parent, $A$ observed). In $(F, B, D, C, E)$, $(F, B, D)$ is active. Indeed, knowing an extremity of a triple does not impact whether it is active or not. Only the center variable is important. Then we have $(B, D, C)$ which is active (v-structure, $D$ observed) and $(D, C, E)$ also. Hence, $(F, B, D, C, E)$ is active and we cannot guarantee that $F \perp E \mid A, D$.
    \end{solution}

    \part $\mathbf{P}(B, E) = \sum_{a, c, d, f} P(a) \mathbf{P}(B \mid a) P(c \mid a) \mathbf{P}(d \mid B, c) \mathbf{P}(E \mid c) \mathbf{P}(f \mid a, B)$

    \begin{solution}
        We notice that
        \begin{equation*}
            \mathbf{P}(A, B, C, D, E, F) = \mathbf{P}(A) \mathbf{P}(B \mid A) \mathbf{P}(C \mid A) \mathbf{P}(D \mid B, C) \mathbf{P}(E \mid C) \mathbf{P}(F \mid A, B)
        \end{equation*}
        corresponds exactly to
        \begin{equation*}
            \mathbf{P}(X_1, \dots, X_n) = \prod_{i = 1}^n \mathbf{P}(X_i \mid \text{parents}(X_i)) ,
        \end{equation*}
        for the considered network, which we know is guaranteed. We also know that
        \begin{equation*}
            \mathbf{P}(B, E) = \sum_{a, c, d, f} \mathbf{P}(a, B, c, d, E, f)
        \end{equation*}
        Hence, the initial statement is always true.
    \end{solution}
\end{parts}

For the same network, use inference by variable elimination to compute $\mathbf{P}(E \mid A = 1, B = 1)$.

\begin{solution}
    We have
    \begin{align*}
        \mathbf{P}(E \mid A = 1, B = 1) & = \frac{1}{Z} \sum_{c, d} \mathbf{P}(A = 1, B = 1, c, d, E) \\
        & = \frac{1}{Z} \sum_{c, d} P(A = 1) P(B = 1 \mid A = 1) P(c \mid A = 1) P(d \mid B = 1, c) \mathbf{P}(E \mid c) \\
        & = \frac{1}{Z} P(A = 1) P(B = 1 \mid A = 1) \\
        & \qquad \sum_{c} P(c \mid A = 1) \mathbf{P}(E \mid c) \sum_d P(d \mid B = 1, c) .
    \end{align*}
    We define the initial factors as
    \begin{align*}
        f_1 & = P(A = 1) \\
        f_2 & = P(B = 1 \mid A = 1) \\
        f_3(C) & = P(C \mid A = 1) \\
        f_4(E, C) & = P(E \mid C) \\
        f_5(C, D) & = P(D \mid B = 1, C)
    \end{align*}
    and the composite factors as
    \begin{align*}
        f_6(C) & = \sum_d f_5(C, d) \\
        f_7(C, E) & = f_3(C) \times f_4(E, C) \times f_6(C) \\
        f_8(E) & = \sum_c f_7(c, E) \\
        f_9(E) & = f_1 \times f_2 \times f_8(E) .
    \end{align*}
    Finally,
    \begin{equation*}
        \mathbf{P}(E \mid A = 1, B = 1) = \frac{1}{Z} f_9(E) = \frac{f_9(E)}{\sum_e f_9(e)}.
    \end{equation*}
\end{solution}

\begin{parts}
    \part Assuming the parameters of the model are known, what is the expected value of $y$ given $x_1$ and $x_2$.

    \begin{solution}
        Our task is to find the expectation
        \begin{equation*}
            \underset{p(y \mid x_1, x_2)}{\E} \sbk{y} = \int y \, p(y \mid x_1, x_2) \d{y} .
        \end{equation*}
        We know that
        \begin{equation*}
            p(y \mid x_1, x_2) = \int p(y \mid x_1, x_2, x_3) \, p(x_3 \mid x_1, x_2) \d{x_3} ,
        \end{equation*}
        where $p(y \mid x_1, x_2, x_3)$ and $p(x_3 \mid x_1, x_2)$ are linear Gaussian distributions given in the statement. Therefore, we have
        \begin{equation*}
            p(y \mid x_1, x_2) = \N\rbk*{y \mid a_3 x_1 + a_4 x_2 + a_5 (a_2 x_2 + \mu_3) + \mu_y, (a_5 \sigma_3)^2 + \sigma_y^2}
        \end{equation*}
        and, by definition of a Gaussian distribution,
        \begin{equation*}
            \underset{p(y \mid x_1, x_2)}{\E} \sbk{y} = a_3 x_1 + (a_4 + a_5 a_2) x_2 + a_5 \mu_3 + \mu_y.
        \end{equation*}
    \end{solution}

    \part Suppose now that the model's parameters are unknown. Given a learning set $d = \cbk{(x_{i,1}, x_{i,2}, y_i)}$ of $N$ independent and identically distributed points, determine the model that best describes $d$.

    \begin{solution}
        \allowdisplaybreaks
        We know that the distribution of $y$ given $x_1$ and $x_2$ takes the form $\N(y \mid w_1 x_1 + w_2 x_2 + b, \sigma^2)$. Then, our task is to find the parameters $h = (w_1, w_2, b, \sigma)$ that maximize the likelihood of $d$, \ie{} the \emph{maximum likelihood estimation} (MLE)
        \begin{align*}
            h_{\text{MLE}} & = \arg \max_h p(d \mid h) \\
            & = \arg \max_h \prod_i p(x_i, y_i \mid h) \\
            & = \arg \max_h \log \prod_i p(x_i, y_i \mid h) \\
            & = \arg \max_h \sum_i \log p(x_i, y_i \mid h) \\
            & = \arg \max_h \sum_i \log p(y_i \mid h, x_i) + \log p(x_i) \\
            & = \arg \max_h \sum_i \log p(y_i \mid h, x_i) \\
            & = \arg \max_h \sum_i \log \sbk*{\frac{1}{\sqrt{2 \pi \sigma^2}} \exp\rbk*{- \frac{(w^T x_i - y_i)^2}{2 \sigma^2}}} \\
            & = \arg \max_h \sum_i - \frac{1}{2} \log (2 \pi \sigma^2) - \frac{(w^T x_i - y_i)^2}{2 \sigma^2} \\
            & = \arg \min_h \log \sigma^2 + \frac{1}{\sigma^2} \frac{1}{N} \sum_i (w^T x_i - y_i)^2 ,
        \end{align*}
        where $x_i = \mat{x_{i,1} & x_{i,2} & 1}^T$ and $w = \mat{w_1 & w_2 & b}^T$. In the last expression, we observe that the summation term is independent from $\sigma$. Therefore,
        \begin{align*}
            w_{\text{MLE}} & = \arg \min_w \sum_i (w^T x_i - y_i)^2 ,
        \end{align*}
        which exactly corresponds to a \emph{linear regression} problem. Then, we find $w_{\text{MLE}}$ by canceling the gradient with respect to $w$, \ie{}
        \begin{align*}
            0 & = \nabla_{\!w} \sum_i (w^T x_i - y_i)^2 \\
            & = \nabla_{\!w} \sum_i (w^T x_i - y_i) (w^T x_i - y_i) \\
            & = \nabla_{\!w} \sum_i (w^T x_i)^2 + y_i^2 - 2 w^T x_i y_i \\
            & = \nabla_{\!w} \rbk*{w^T X^T X w + Y^T Y - 2 w^T X^T Y} \\
            & = 2 X^T X w - 2 X^T Y
        \end{align*}
        where $X = (x_i^T) \in \R^{N \times 3}$ and $Y = (y_i) \in \R^{N}$. Finally, we have
        \begin{alignat*}{2}
            && 0 & = X^T X w_{\text{MLE}} - X^T Y \\
            \Leftrightarrow \quad && w_{\text{MLE}} & = (X^T X)^{-1} X^T Y .
        \end{alignat*}
        Afterwards, we find $\sigma_{\text{MLE}}$ such that
        \begin{align*}
            \sigma_{\text{MLE}} & = \arg \min_\sigma \log \sigma^2 + \frac{\text{MSE}}{\sigma^2} \\
            & = \sqrt{\text{MSE}},
        \end{align*}
        where $\text{MSE}$ denotes the \emph{mean squared error}
        \begin{equation*}
            \frac{1}{N} \sum_i ({w_{\text{MLE}}}^T x_i - y_i)^2 .
        \end{equation*}
    \end{solution}
\end{parts}

\titledquestion{Heteroscedastic linear regression}

In linear regression with Gaussian noise of constant variance, $y_i \sim \N(w^T x_i, \sigma^2)$, the maximum likelihood estimate of the weights is $w_{\text{MLE}} = (X^T X)^{-1} X^T Y$, where the rows of $X$ are the inputs $x_i^T$ and $Y = (y_i)$. What does this estimate become if the noise is different for each sample, \ie{} $y_i \sim \N(w^T x_i, \sigma_i^2)$ with known variances $\sigma_i^2$?

\begin{solution}
    The log-likelihood of the weights is
    \begin{equation*}
        L(w) = \sum_i \log \N(y_i \mid w^T x_i, \sigma_i^2) = - \sum_i \frac{(w^T x_i - y_i)^2}{2 \sigma_i^2} + \text{const},
    \end{equation*}
    so that maximizing it minimizes a weighted sum of squares, each sample weighted by its precision $1 / \sigma_i^2$. With $W = \diag(1 / \sigma_1^2, \ldots, 1 / \sigma_N^2)$,
    \begin{equation*}
        \nabla_{\!w} L(w) = - X^T W (X w - Y) = 0
        \quad \Leftrightarrow \quad
        w_{\text{MLE}} = (X^T W X)^{-1} X^T W Y .
    \end{equation*}
    Noisy samples count less. With equal variances, $W$ is proportional to the identity and the estimate of the homoscedastic case is recovered.
\end{solution}
