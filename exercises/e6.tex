\begin{recap}
    \topic{Problem statement}
    Let $\mathbf{d} \sim p(\mathbf{x}, y)$ be a dataset of $N$ example input-output pairs
    \[
        \mathbf{d} = \{(\mathbf{x}_1, y_1), (\mathbf{x}_2, y_2), \ldots, (\mathbf{x}_N, y_N)\},
    \]
    where $\mathbf{x}_i \in \mathbb{R}^d$ are $d$-dimensional vectors representing the input values and $y_i \in \mathcal{Y}$ are the corresponding output values. From this data, we want to identify a probabilistic model $p_\theta(y \mid \mathbf{x})$ that best explains the data. The problem is a \term{regression} problem if $y \in \mathbb{R}$ and a \term{classification} problem if $y \in \{0, 1, \ldots, C-1\}$.

    \topic{Linear regression}
    Linear regression models $p(y \mid \mathbf{x}) = \mathcal{N}(y \mid \mathbf{w}^T \mathbf{x} + b, \sigma^2)$. Its maximum likelihood estimate minimizes the sum of squared errors $\sum_{i=1}^N (y_i - (\mathbf{w}^T \mathbf{x}_i + b))^2$. Appending a constant feature 1 to $\mathbf{x}$ to absorb $b$ into $\mathbf{w}$, it is $\mathbf{w}^* = (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T \mathbf{y}$, where $\mathbf{X}$ stacks the inputs $\mathbf{x}_i^T$ and $\mathbf{y}$ the outputs $y_i$.

    \topic{Logistic regression}
    Logistic regression models $P(Y = 1 \mid \mathbf{x}) = \sigma(\mathbf{w}^T \mathbf{x} + b)$, with the sigmoid $\sigma(z) = 1 / (1 + \exp(-z))$. Its maximum likelihood estimate minimizes the \term{cross-entropy} $\sum_{i=1}^N -y_i \log \sigma(\mathbf{w}^T \mathbf{x}_i + b) - (1 - y_i) \log(1 - \sigma(\mathbf{w}^T \mathbf{x}_i + b))$, which has no closed-form minimizer.

    \topic{Gradient descent}
    To minimize a loss function $\mathcal{L}(\theta)$, \term{gradient descent} updates the model parameters iteratively using the update rule
    \[
        \theta_{t+1} = \theta_t - \alpha \nabla_\theta \mathcal{L}(\theta_t),
    \]
    where $\theta_0$ are the initial parameters of the model and $\alpha$ is the \term{learning rate}. When $\mathcal{L}(\theta) = \frac{1}{N} \sum_{i} \ell(y_i, f(\mathbf{x}_i; \theta))$, \term{stochastic gradient descent} replaces the gradient by its average over a random minibatch of the pairs.

    \topic{Multilayer perceptron}
    A \term{shallow network} with input $\mathbf{x} \in \mathbb{R}^{d_{\text{in}}}$, $q$ hidden units and one output computes
    \[
        h_j = \sigma\Big(\sum_{i=0}^{d_{\text{in}}-1} w_{ji} x_i + b_j\Big) \quad \text{for } j = 0, \ldots, q-1, \qquad y = \sum_{j=0}^{q-1} v_j h_j + c,
    \]
    where $\sigma$ is an activation function. In matrix form, a hidden layer computes
    \[
        \mathbf{h} = \sigma(\mathbf{W} \mathbf{x} + \mathbf{b}),
    \]
    where $\mathbf{W} \in \mathbb{R}^{q \times d_{\text{in}}}$ is the \term{weight matrix}, of entries $w_{ji}$, and $\mathbf{b} \in \mathbb{R}^q$ is the \term{bias vector}. Layers composed in series form a deep network with $K$ hidden layers,
    \begin{align*}
        \mathbf{h}_0 & = \mathbf{x} \\
        \mathbf{h}_k & = \sigma(\mathbf{W}_k \mathbf{h}_{k-1} + \mathbf{b}_k) \quad \text{for } k = 1, \ldots, K \\
        \mathbf{y} & = \sigma_\text{out}(\mathbf{W}_y \mathbf{h}_K + \mathbf{b}_y),
    \end{align*}
    where $\mathbf{W}_k \in \mathbb{R}^{q_k \times q_{k-1}}$ is the weight matrix and $\mathbf{b}_k \in \mathbb{R}^{q_k}$ is the bias vector of the $k$-th layer. The resulting function $f(\mathbf{x}; \theta)$, where $\theta$ denotes the set of all model parameters, is a \term{multilayer perceptron}. With ReLU units, it is piecewise linear: a shallow network with one input has at most $q + 1$ linear regions, a deep network with $K$ layers of $q$ units up to $(q + 1)^K$.

    \topic{Universal approximation theorem}
    Let $\sigma : \mathbb{R} \to \mathbb{R}$ be continuous and not a polynomial. For every compact set $K \subset \mathbb{R}^{d_\text{in}}$, every continuous function $f : K \to \mathbb{R}$ and every $\epsilon > 0$, there exist $q$, $w_{ji}$, $b_j$, $v_j$ and $c$ such that
    \[
        \sup_{\mathbf{x} \in K} \Big| f(\mathbf{x}) - \Big( \sum_{j=0}^{q-1} v_j \, \sigma\Big( \sum_{i=0}^{d_\text{in}-1} w_{ji} x_i + b_j \Big) + c \Big) \Big| < \epsilon:
    \]
    a shallow network with enough hidden units approximates any continuous function on a closed and bounded domain arbitrarily well.

    \topic{Activation functions}
    Activation functions $\sigma : \mathbb{R} \to \mathbb{R}$ introduce non-linearities in the network. Common activation functions include
    \begin{itemize}
        \item the sigmoid $\sigma(x) = \frac{1}{1 + \exp(-x)}$, which maps inputs to the range $(0, 1)$;
        \item the hyperbolic tangent $\sigma(x) = \tanh(x) = \frac{\exp(x) - \exp(-x)}{\exp(x) + \exp(-x)}$, which maps inputs to the range $(-1, 1)$;
        \item the rectified linear unit $\sigma(x) = \ReLU(x) = \max(0, x)$.
    \end{itemize}

    \topic{Output layers}
    The output layer $\mathbf{y} = \sigma_\text{out}(\mathbf{W}_y \mathbf{h}_K + \mathbf{b}_y)$ depends on the task:
    \begin{itemize}
        \item for regression, its width is the dimensionality $d_{\text{out}}$ of the output and $\sigma_\text{out}$ is the identity;
        \item for binary classification, its width is $1$ and $\sigma_\text{out}$ is the sigmoid, whose output is $P(Y = 1 \mid \mathbf{x})$;
        \item for classification into $C$ classes, its width is $C$ and $\sigma_\text{out}$ is the \term{softmax}
        \[
            \mathrm{Softmax}(\mathbf{z})_i = \frac{\exp(z_i)}{\sum_{i'=0}^{C-1} \exp(z_{i'})} \quad \text{for } i = 0, \ldots, C-1,
        \]
        so that $\mathbf{y}$ is a vector of $C$ probabilities $P(Y = i \mid \mathbf{x})$ summing to 1.
    \end{itemize}

    \topic{Loss functions}
    The parameters $\theta$ of a deep network $f(\mathbf{x}; \theta)$ are learned by minimizing a loss function $\mathcal{L}(\theta)$ over the dataset $\mathbf{d} = \{(\mathbf{x}_i, \mathbf{y}_i)\}$ of $N$ input-output pairs. The loss function is derived from the likelihood:
    \begin{itemize}
        \item for regression, assuming a Gaussian likelihood, the loss is the \term{mean squared error}
        \[
            \mathcal{L}(\theta) = \frac{1}{N} \sum_{i=1}^N \lVert \mathbf{y}_i - f(\mathbf{x}_i; \theta) \rVert^2;
        \]
        \item for classification, assuming a categorical likelihood, the loss is the \term{cross-entropy}, the negative log-probability of the observed class,
        \[
            \mathcal{L}(\theta) = -\frac{1}{N} \sum_{i=1}^N \log f_{y_i}(\mathbf{x}_i; \theta).
        \]
    \end{itemize}

    \topic{Chain rule}
    The gradient of the loss is obtained by the chain rule. For example, for $h_1 = \sigma(w_1 x + b_1)$, $\hat{y} = w_2 h_1 + b_2$ and $\ell = (y - \hat{y})^2$, the chain $w_1 \to h_1 \to \hat{y} \to \ell$ gives
    \[
        \frac{\partial \ell}{\partial w_1} = \frac{\partial \ell}{\partial \hat{y}} \frac{\partial \hat{y}}{\partial h_1} \frac{\partial h_1}{\partial w_1} = -2(y - \hat{y}) \, w_2 \, \sigma'(w_1 x + b_1) \, x.
    \]
    On the \term{computation graph} of the network (variables and the operations that produce them), the \term{forward pass} computes every intermediate value from the input to the loss; the \term{backward pass} then multiplies the local derivatives from the loss back to each parameter, reusing the values of the forward pass. \term{Automatic differentiation} implements this procedure for any program.

    \topic{Generalization}
    We want a small loss on new pairs drawn from $p(\mathbf{x}, y)$, not on the pairs of $\mathbf{d}$. An honest estimate comes from a \term{test set} $\mathbf{d}_\text{test}$, held out from training. A model too small for the data \term{underfits} (both losses stay high); a model large enough to memorize $\mathbf{d}$ \term{overfits} (the training loss decreases while the test loss increases). \term{Hyperparameters} (width, depth, learning rate, minibatch size, number of epochs) are chosen on a \term{validation set}, also held out from training; the test set is used once, at the end.

    \topic{Convolution}
    For the 1d input $\mathbf{x} \in \mathbb{R}^W$ and the convolutional kernel $\omega \in \mathbb{R}^w$, the discrete \term{convolution} $\mathbf{x} \circledast \omega$ is a vector of size $W - w + 1$ such that
    \[
        (\mathbf{x} \circledast \omega)[i] = \sum_{m=0}^{w-1} \mathbf{x}_{m+i} \omega_m.
    \]
    Technically, $\circledast$ is the cross-correlation: the kernel is not flipped. Padding the input with $p$ zeros on each side results in an output of size $W - w + 1 + 2p$. For the 3d input tensor $\mathbf{x} \in \mathbb{R}^{C \times H \times W}$ with $C$ channels and the 3d convolutional kernel $\omega \in \mathbb{R}^{C \times h \times w}$, the 2d convolution $\mathbf{x} \circledast \omega$ is a 2d tensor of size $(H - h + 1) \times (W - w + 1)$ such that
    \[
        (\mathbf{x} \circledast \omega)[j, i] = \sum_{c=0}^{C-1} \sum_{n=0}^{h-1} \sum_{m=0}^{w-1} \mathbf{x}_{c, n+j, m+i} \omega_{c, n, m}.
    \]

    \topic{Convolutional neural networks}
    A \term{convolutional layer} is defined by a set of $K$ kernels $\omega_k$ of size $C \times h \times w$. It applies the 2d convolution to the input tensor $\mathbf{x}$ of size $C \times H \times W$ to produce a set of $K$ feature maps $\mathbf{o}_k$, the $K$ channels of its output. With a pooling area of size $h \times w$, \term{max-pooling} maps a tensor $\mathbf{x} \in \mathbb{R}^{C \times (rh) \times (sw)}$ to a tensor $\mathbf{o} \in \mathbb{R}^{C \times r \times s}$ such that
    \[
        \mathbf{o}_{c,j,i} = \max_{n < h, m < w} \mathbf{x}_{c,hj+n,wi+m}.
    \]
    The \term{receptive field} of a value is the patch of the input it depends on: a $w \times w$ kernel sees $w \times w$ values, and each further convolutional or pooling layer widens it. A \term{convolutional neural network} (CNN) uses convolutional layers, pooling layers and fully connected layers.
\end{recap}

\titledquestion{Escape game \source{Exam, January 2022}}

A new virtual escape game came out, and you decide to play it. You arrive in a $5 \times 5$ grid world where each cell $(x, y)$ is a room with doors leading to the adjacent rooms. The game's goal is to reach the exit room as fast as possible, but its position is unknown. Furthermore, some regions of the world are full of riddles, and crossing rooms in these regions takes longer. Fortunately, a leaderboard with the players' best times is provided, starting from a few different rooms. Due to rounding errors, you assume that the best times reported in the leaderboard are measurements affected by additive Gaussian noise $\N(0, 1)$.

\begin{center}
    \begin{tabular}{ccc}
        \toprule
        $i$ & Starting room & Measured best time \\
        \midrule
        1 & $(4, 5)$ & $2.0$ \\
        2 & $(5, 3)$ & $3.5$ \\
        3 & $(3, 3)$ & $4.5$ \\
        4 & $(4, 1)$ & $7.0$ \\
        5 & $(1, 2)$ & $8.5$ \\
        \bottomrule
    \end{tabular}
\end{center}

From the leaderboard, you wish to learn a heuristic approximating the best time to get to the exit, starting from room $(x, y)$. You decide to use a small neural network as approximator, described by the following parametric function,
\begin{align*}
    h(x, y; \phi) & = \text{ReLU}(x w_1 + y w_2 + w_3) + \text{ReLU}(x w_4 + y w_5 + w_6) \\
    \text{ReLU}(x) & = \max(x, 0) ,
\end{align*}
where $\phi = (w_1, w_2, w_3, w_4, w_5, w_6)$ is the set of parameters/weights of the neural network.

\begin{parts}
    \part Among the following sets of parameters ($A$, $B$ or $C$), which one would you use? Justify your answer.

    \begin{center}
        \begin{tabular}{ccccccc}
            \toprule
            Set & $w_1$ & $w_2$ & $w_3$ & $w_4$ & $w_5$ & $w_6$ \\
            \midrule
            $\phi_A$ & $-1.5$ & $1$ & $4$ & $1$ & $-1.5$ & $6$ \\
            $\phi_B$ & $-1$ & $1.5$ & $3$ & $0$ & $-1$ & $4$ \\
            $\phi_C$ & $-2$ & $0.5$ & $4.5$ & $1.5$ & $0$ & $5$ \\
            \bottomrule
        \end{tabular}
    \end{center}

    \begin{solution}
        A set of parameters is better than another if it explains the data better, that is if the likelihood is higher. Our task is to find the set of parameters that maximizes the likelihood under the assumed probability model. In our case, the data $d$ is the leaderboard and consists of independent position-time tuples $(x_i, y_i, t_i)$. From the statement, we gather that the (best) time $t$ is a function of the starting position $(x, y)$, approximated by a neural network $h(x, y; \phi)$. We also learn that our time measurements are affected by Gaussian noise.
        \begin{align*}
            \phi_{\text{MLE}} & = \arg\max_\phi p(\mathbf{d} \mid \phi) \\
            & = \arg\max_\phi \prod_{i=1}^5 p(x_i, y_i, t_i \mid \phi) \\
            & = \arg\max_\phi \prod_{i=1}^5 p(t_i \mid x_i, y_i, \phi) \, p(x_i, y_i) \\
            & = \arg\max_\phi \prod_{i=1}^5 \N(t_i; h(x_i, y_i; \phi), 1) \\
            & = \arg\max_\phi \sum_{i=1}^5 \log \sbk*{\frac{1}{\sqrt{2 \pi}} \exp \rbk*{-\frac{(t_i - h(x_i, y_i; \phi))^2}{2}}} \\
            & = \arg\min_\phi \sum_{i=1}^5 (t_i - h(x_i, y_i; \phi))^2
        \end{align*}
        Now that we have an objective $L(\phi)$ to minimize, we can evaluate it for each of the proposed sets and select the best one.
        \begin{align*}
            L(\phi_A) & = \sum_{i=1}^5 (t_i - h(x_i, y_i; \phi_A))^2 = (2.0 - 5.5)^2 + (3.5 - 6.5)^2 + \dots = 29.75 \\
            L(\phi_B) & = 35.75 \\
            L(\phi_C) & = 205.25
        \end{align*}
        The best set is $\phi_A$.
    \end{solution}

    \part You now assume a Gaussian prior $\N(0, 1)$ on each parameter. Which set of parameters in the table above would you now choose? Justify your answer.

    \begin{solution}
        Now that we hold a prior belief on the parameters, our task is to find the most likely set of parameters given the data, that is maximum a posteriori (MAP) estimation.
        \begin{align*}
            \phi_{\text{MAP}} & = \arg\max_\phi p(\phi \mid \mathbf{d}) \\
            & = \arg\max_\phi p(\mathbf{d} \mid \phi) p(\phi) \\
            & = \arg\max_\phi \prod_{i=1}^5 p(x_i, y_i, t_i \mid \phi) \prod_{j=1}^6 p(w_j) \\
            & = \arg\max_\phi \prod_{i=1}^5 \N(t_i; h(x_i, y_i; \phi), 1) \prod_{j=1}^6 \N(w_j; 0, 1) \\
            & = \arg\min_\phi \sum_{i=1}^5 (t_i - h(x_i, y_i; \phi))^2 + \underbrace{\sum_{j=1}^6 (w_j)^2}_{\norm{\phi}^2}
        \end{align*}
        Here again, we have an objective $L'(\phi)$ to minimize, which we evaluate for each of the proposed sets.
        \begin{align*}
            L'(\phi_A) & = L(\phi_A) + \sum_{j=1}^6 (w_j)^2 = 29.75 + (-1.5)^2 + (1)^2 + \dots = 88.25 \\
            L'(\phi_B) & = 65.0 \\
            L'(\phi_C) & = 257.0
        \end{align*}
        The best set is $\phi_B$.
    \end{solution}

    \part Discuss the procedure you would implement on a computer to find the optimal set of parameters, had the table above not been provided.

    \begin{solution}
        If the table is not provided, it is intractable to evaluate and compare all possible sets of parameters. However, we still want to minimize the MLE (or MAP) objective $L(\phi)$. We can improve a set of parameters $\phi$ by following the opposite of the objective's gradient $\nabla_{\! \phi} L(\phi)$, \ie{} performing \emph{gradient descent} steps
        \begin{equation*}
            \phi \gets \phi - \alpha \nabla_{\! \phi} L(\phi) ,
        \end{equation*}
        where $\alpha$ is the learning rate.
    \end{solution}
\end{parts}

\titledquestion{Predict the leaderboard score}

A leaderboard ranks language models by an overall score. The Bayesian network below represents how this score is computed. In this model, $x_1$, $x_2$ and $x_3$ respectively denote the scores obtained by a model on a validation set, on a public benchmark and on a held-out test set. The three evaluations share the same harness, and a model that does well on one tends to do well on the next. In particular, $x_2 \sim \N(a_1 x_1 + \mu_2, \sigma_2^2)$ and $x_3 \sim \N(a_2 x_2 + \mu_3, \sigma_3^2)$. Finally, $y \sim \N(a_3 x_1 + a_4 x_2 + a_5 x_3 + \mu_y, \sigma_y^2)$ stands for the overall score, which is a linear combination of the three scores plus some Gaussian noise due to rounding errors. Answer the following questions about this model.

\begin{center}
    \begin{tikzpicture}[bn diagram]
        \node[random variable] (X1) at (0, 0) {$x_1$};
        \node[random variable] (X2) at (2, 0) {$x_2$};
        \node[random variable] (X3) at (4, 0) {$x_3$};
        \node[random variable] (Y) at (2, -1.5) {$y$};
        \draw[dependency] (X1) -- (X2);
        \draw[dependency] (X2) -- (X3);
        \draw[dependency] (X1) -- (Y);
        \draw[dependency] (X2) -- (Y);
        \draw[dependency] (X3) -- (Y);
    \end{tikzpicture}
\end{center}

\begin{parts}
    \part Assuming the parameters of the model are known, what is the expected value of $y$ given $x_1$ and $x_2$?

    \begin{solution}
        Our task is to find the expectation
        \begin{equation*}
            \underset{p(y \mid x_1, x_2)}{\E} \sbk{y} = \int y \, p(y \mid x_1, x_2) \d{y} .
        \end{equation*}
        We know that
        \begin{equation*}
            p(y \mid x_1, x_2) = \int p(y \mid x_1, x_2, x_3) \, p(x_3 \mid x_1, x_2) \d{x_3} ,
        \end{equation*}
        where $p(y \mid x_1, x_2, x_3)$ is given in the statement and $p(x_3 \mid x_1, x_2) = p(x_3 \mid x_2)$, since $x_3 \perp x_1 \mid x_2$. Both are linear Gaussian distributions. Therefore, we have
        \begin{equation*}
            p(y \mid x_1, x_2) = \N\rbk*{y \mid a_3 x_1 + a_4 x_2 + a_5 (a_2 x_2 + \mu_3) + \mu_y, (a_5 \sigma_3)^2 + \sigma_y^2}
        \end{equation*}
        and, by definition of a Gaussian distribution,
        \begin{equation*}
            \underset{p(y \mid x_1, x_2)}{\E} \sbk{y} = a_3 x_1 + (a_4 + a_5 a_2) x_2 + a_5 \mu_3 + \mu_y.
        \end{equation*}
    \end{solution}

    \part Suppose now that the parameters are unknown, and that the scores $x_3$ on the held-out test set are not observed. Given a learning set $\mathbf{d} = \cbk{(x_{i,1}, x_{i,2}, y_i)}$ of $N$ independent and identically distributed points, derive the maximum likelihood estimate of the parameters of $p(y \mid x_1, x_2)$.

    \begin{solution}
        \allowdisplaybreaks
        We know that the distribution of $y$ given $x_1$ and $x_2$ takes the form $\N(y \mid w_1 x_1 + w_2 x_2 + b, \sigma^2)$. Then, our task is to find the parameters $h = (w_1, w_2, b, \sigma)$ that maximize the likelihood of $d$, \ie{} the \emph{maximum likelihood estimation} (MLE)
        \begin{align*}
            h_{\text{MLE}} & = \arg \max_h p(\mathbf{d} \mid h) \\
            & = \arg \max_h \prod_i p(x_i, y_i \mid h) \\
            & = \arg \max_h \log \prod_i p(x_i, y_i \mid h) \\
            & = \arg \max_h \sum_i \log p(x_i, y_i \mid h) \\
            & = \arg \max_h \sum_i \log p(y_i \mid h, x_i) + \log p(x_i) \\
            & = \arg \max_h \sum_i \log p(y_i \mid h, x_i) \\
            & = \arg \max_h \sum_i \log \sbk*{\frac{1}{\sqrt{2 \pi \sigma^2}} \exp\rbk*{- \frac{(w^T x_i - y_i)^2}{2 \sigma^2}}} \\
            & = \arg \max_h \sum_i - \frac{1}{2} \log (2 \pi \sigma^2) - \frac{(w^T x_i - y_i)^2}{2 \sigma^2} \\
            & = \arg \min_h \log \sigma^2 + \frac{1}{\sigma^2} \frac{1}{N} \sum_i (w^T x_i - y_i)^2 ,
        \end{align*}
        where $x_i = \mat{x_{i,1} & x_{i,2} & 1}^T$ and $w = \mat{w_1 & w_2 & b}^T$. In the last expression, we observe that the minimizer of the summation term is independent of $\sigma$. Therefore,
        \begin{align*}
            w_{\text{MLE}} & = \arg \min_w \sum_i (w^T x_i - y_i)^2 ,
        \end{align*}
        which exactly corresponds to a \emph{linear regression} problem. Then, we find $w_{\text{MLE}}$ by canceling the gradient with respect to $w$, \ie{}
        \begin{align*}
            0 & = \nabla_{\!w} \sum_i (w^T x_i - y_i)^2 \\
            & = \nabla_{\!w} \sum_i (w^T x_i - y_i) (w^T x_i - y_i) \\
            & = \nabla_{\!w} \sum_i \rbk*{(w^T x_i)^2 + y_i^2 - 2 w^T x_i y_i} \\
            & = \nabla_{\!w} \rbk*{w^T X^T X w + Y^T Y - 2 w^T X^T Y} \\
            & = 2 X^T X w - 2 X^T Y
        \end{align*}
        where $X = (x_i^T) \in \R^{N \times 3}$ and $Y = (y_i) \in \R^{N}$. Finally, we have
        \begin{alignat*}{2}
            && 0 & = X^T X w_{\text{MLE}} - X^T Y \\
            \Leftrightarrow \quad && w_{\text{MLE}} & = (X^T X)^{-1} X^T Y .
        \end{alignat*}
        Afterwards, we find $\sigma_{\text{MLE}}$ such that
        \begin{align*}
            \sigma_{\text{MLE}} & = \arg \min_\sigma \log \sigma^2 + \frac{\text{MSE}}{\sigma^2} \\
            & = \sqrt{\text{MSE}},
        \end{align*}
        where $\text{MSE}$ denotes the \emph{mean squared error}
        \begin{equation*}
            \frac{1}{N} \sum_i ({w_{\text{MLE}}}^T x_i - y_i)^2 .
        \end{equation*}
        By (a), the fitted parameters estimate combinations of those of the network, $w_1 = a_3$, $w_2 = a_4 + a_5 a_2$, $b = a_5 \mu_3 + \mu_y$ and $\sigma^2 = a_5^2 \sigma_3^2 + \sigma_y^2$. Without observations of $x_3$, the individual parameters $a_2, a_4, a_5, \mu_3, \mu_y, \sigma_3, \sigma_y$ cannot be recovered separately.
    \end{solution}
\end{parts}

\titledquestion{Heteroscedastic linear regression}

In linear regression with Gaussian noise of constant variance, $y_i \sim \N(w^T x_i, \sigma^2)$, the maximum likelihood estimate of the weights is $w_{\text{MLE}} = (X^T X)^{-1} X^T Y$, where the rows of $X$ are the inputs $x_i^T$ and $Y = (y_i)$. What does this estimate become if the noise is different for each sample, \ie{} $y_i \sim \N(w^T x_i, \sigma_i^2)$ with known variances $\sigma_i^2$?

\begin{solution}
    The log-likelihood of the weights is
    \begin{equation*}
        L(w) = \sum_i \log \N(y_i \mid w^T x_i, \sigma_i^2) = - \sum_i \frac{(w^T x_i - y_i)^2}{2 \sigma_i^2} + \text{const},
    \end{equation*}
    so that maximizing it minimizes a weighted sum of squares, each sample weighted by its precision $1 / \sigma_i^2$. With $W = \diag(1 / \sigma_1^2, \ldots, 1 / \sigma_N^2)$,
    \begin{equation*}
        \nabla_{\!w} L(w) = - X^T W (X w - Y) = 0
        \quad \Leftrightarrow \quad
        w_{\text{MLE}} = (X^T W X)^{-1} X^T W Y .
    \end{equation*}
    Noisy samples count less. With equal variances, $W$ is proportional to the identity and the estimate of the homoscedastic case is recovered.
\end{solution}

\titledquestion{Binary classification \source{CS188, Spring 2024}}

Consider the following simple neural network for binary classification. Here $x$ is a single real-valued input feature with an associated class $y$ (0 or 1). There are two weight parameters $w_1$ and $w_2$, and non-linearity functions $g_1$ and $g_2$. The network will output a value $a_2$ between 0 and 1, representing the probability of being in class 1. We will be using a loss function $\mathcal{L}$ to compare the prediction $a_2$ with the true class $y$.

\begin{center}
    \includegraphics[width=\linewidth]{figures/classif_2.png}
\end{center}

\begin{parts}
    \part Write down the forward pass on this network, writing the output values for each node $z_1$, $a_1$, $z_2$ and $a_2$ in terms of the node's input values.

    \begin{solution}
        \begin{align*}
            z_1 &= w_1 x \\
            a_1 &= g_1(z_1) = g_1(w_1 x) \\
            z_2 &= w_2 a_1 = w_2 g_1(w_1 x) \\
            a_2 &= g_2(z_2) = g_2(w_2 g_1(w_1 x))
        \end{align*}
    \end{solution}

    \part Derive the arguments of the loss $\mathcal{L}(a_2, y)$ in terms of the input $x$, weights $w_i$, and activation functions $g_i$.

    \begin{solution}
        The argument $a_2$ has been derived in the previous question. We have
        \[
            a_2 = g_2(w_2 g_1(w_1 x)).
        \]
        For the other input $y$, the latter is given as input of the problem. Indeed, in supervised learning, we assume access to a training dataset made of input-output pairs $\{(x_i, y_i)\}^N$ where $y_i$ acts as a target for the network.
    \end{solution}

    \part Using the chain rule, differentiate $\mathcal{L}$ w.r.t. $w_2$. Write your expression as a product of partial derivatives at each node: \ie{} the partial derivative of the node's output with respect to its inputs. (Hint: the series of expressions you wrote in (a) will be helpful; you may use any of those variables.)

    \begin{solution}
        \[
            \dfrac{\partial \mathcal{L}}{\partial w_2} = \dfrac{\partial \mathcal{L}}{\partial a_2} \dfrac{\partial a_2}{\partial z_2} \dfrac{\partial z_2}{\partial w_2}
        \]
    \end{solution}

    \part Motivate a choice for the output activation function $g_2$ given the nature of the problem. Motivate also a loss function. Give the expression of the gradients of both functions w.r.t. to their inputs.

    \begin{solution}
        From the statement
        \begin{enumerate}[label=(\alph*)]
            \item we know that we are in a classification problem (specifically, a binary classification),
            \item we know that the output $a_2$ should be between 0 and 1 and represents the probability $p(Y = 1 \mid x)$.
        \end{enumerate}
        From (b), we have $a_2: \mathbb{R} \mapsto [0,1]$. A typical choice that fulfill this requirement is the sigmoid function $\sigma(z) = \dfrac{1}{1 + e^{-z}}$. The gradient of the sigmoid w.r.t. its input is
        \[
            \dfrac{\partial \sigma(z)}{\partial z} = \dfrac{e^{-z}}{(1 + e^{-z})^2} = \sigma(z)\sigma(-z) = \sigma(z)(1-\sigma(z)).
        \]
        For the loss function, either we remember the cross-entropy loss seen in the theoretical lecture, or we compute it from maximum likelihood estimation.

        The likelihood of a pair $(x, y)$ can be derived from our model as
        \[
            \begin{cases}
                \text{if } y = 1\ \text{ then }\ P(Y = y \mid x) = \text{NN}_{\theta}(x) \\
                \text{if } y = 0\ \text{ then }\ P(Y = y \mid x) = 1 -\text{NN}_{\theta}(x)
            \end{cases}
        \]
        where $\text{NN}_{\theta}$ is the neural network parameterized by $\theta = [w_1, w_2]$. Denoting $\text{NN}_{\theta}(x) = a_2$, the likelihood expression for one pair can be reformulated as
        \[
            L(x, y; \theta) = y\log a_2 + (1-y) \log (1 - a_2).
        \]
        Since we want to maximize the likelihood, but minimize the loss, we can set the latter to
        \[
            \mathcal{L}(a_2, y) = -y\log a_2 - (1-y) \log (1 - a_2)
        \]
        which corresponds to the binary cross-entropy loss. The gradient of the loss w.r.t. $a_2$ is
        \[
            \dfrac{\partial \mathcal{L}(a_2, y)}{\partial a_2} = -\dfrac{y}{a_2} + \dfrac{(1-y)}{(1-a_2)} = \dfrac{a_2 - y}{a_2(1- a_2)}.
        \]
    \end{solution}

    \part We set the loss function to the binary cross-entropy
    \[
        \mathcal{L}(a_2, y) = -y\log a_2 - (1-y) \log (1 - a_2),
    \]
    and $g_1$ and $g_2$ are both sigmoid functions $\sigma(z) = \frac{1}{1 + e^{-z}}$. If you made this choice in the previous question, use your results to get the expression of $\frac{\partial \mathcal{L}}{\partial w_2}$. Otherwise, start by expressing $\frac{\partial \sigma(z)}{\partial z}$ and $\frac{\partial \mathcal{L}}{\partial a_2}$.

    \begin{solution}
        We first miss $\frac{\partial z_2}{\partial w_2}$ which is
        \[
            \frac{\partial z_2}{\partial w_2} = \frac{\partial}{\partial w_2} (w_2a_1) = a_1.
        \]
        Plugging all the partial derivatives into the expression found in (c), we have
        \begin{align*}
            \dfrac{\partial \mathcal{L}}{\partial w_2} &= \dfrac{a_2 - y}{a_2 (1 - a_2)} g_2(z_2)(1 - g_2(z_2)) a_1 \\
            &= \dfrac{a_2 - y}{a_2 (1 - a_2)} a_2(1 - a_2) a_1 \\
            &= (a_2 - y) a_1
        \end{align*}
    \end{solution}

    \part Now use the chain rule to express $\frac{\partial \mathcal{L}}{\partial w_1}$ as a product of partial derivatives at each node of interest.

    \begin{solution}
        \[
            \frac{\partial \mathcal{L}}{\partial w_1} = \dfrac{\partial \mathcal{L}}{\partial a_2} \dfrac{\partial a_2}{\partial z_2} \dfrac{\partial z_2}{\partial a_1} \dfrac{\partial a_1}{\partial z_1} \dfrac{\partial z_1}{\partial w_1}
        \]
    \end{solution}

    \part Finally, write $\dfrac{\partial \mathcal{L}}{\partial w_1}$ in terms of $x$, $y$, $w_i$, $a_i$, $z_i$.

    \begin{solution}
        We first need to compute
        \[
            \frac{\partial a_1}{\partial z_1} = g_1(z_1)(1 - g_1(z_1)) = a_1(1-a_1)
        \]
        similarly to $\frac{\partial a_2}{\partial z_2}$, and
        \[
            \frac{\partial z_1}{\partial w_1} = \frac{\partial}{\partial w_1} (w_1x) = x.
        \]
        Then, we get
        \[
            \frac{\partial \mathcal{L}}{\partial w_1} = (a_2 - y)w_2 a_1(1-a_1)x.
        \]
    \end{solution}

    \part What is the gradient descent update for $w_1$ with learning rate $\alpha$ in terms of the values computed above?

    \begin{solution}
        \[
            w_1 \leftarrow w_1 - \alpha (a_2 - y)w_2 a_1(1-a_1)x.
        \]
        You see here the importance of the forward pass. Iteratively, when receiving the input $x$, we compute all the intermediate activations (here, $a_1$ and $a_2$) and the loss (given $y$). Then, we propagate backward (from the output to the input) the values following the chain rule for the gradient of the loss w.r.t. parameters of interest. Finally, we update the parameters using the computed gradient.
    \end{solution}
\end{parts}

You would like to formulate a supervised learning problem in order to develop a deep learning model suited to a given task.

You are interested in three different tasks:
\begin{enumerate}
    \item Assign a beauty score (a real-valued quantity) to a given image.
    \item Determine whether the image contains a cat or a dog.
    \item Determine whether each pixel of an image belongs to a cat, a dog or the background.
\end{enumerate}

For each problem, you want to identify a probabilistic model in the form $p_\theta(y \mid x)$ from a dataset that you will construct:
\[
    \mathbf{d} = \{(x_1, y_1), (x_2, y_2), \ldots, (x_N, y_N)\}.
\]

\begin{parts}
    \part For each task, define the random variables $X$ and $Y$, specify their respective dimensions, and explain what $\theta$ represents.

    \begin{solution}
        For each task, $\theta$ represents the parameters of the model being developed. $X$ is the random variable of the image given as input to the model, $X \in \mathbb{R}^{3 \times 32 \times 32}$.

        For the first task, $Y$ is the random variable of the beauty score associated with an image, $Y \in \mathbb{R}$.

        For the second task, $Y$ is a binary random variable indicating whether the image contains a dog (let's say $Y=1$ in this case) or a cat ($Y=0$): $Y \in \{0, 1\}$.

        For the third case, $Y$ is the $32 \times 32$ array of categorical random variables $P_{u,v}$, one for each pixel of the image $X$:
        \[
            Y = (P_{u,v})_{0 \leq u, v \leq 31}.
        \]
        Each $P_{u,v}$ indicates whether pixel $(u,v)$ belongs to a cat, a dog, or the background: $P_{u,v} \in \{\text{cat}, \text{dog}, \text{background}\}$.
    \end{solution}

    \uplevel{Once the task has been formulated, you construct the dataset $\mathbf{d}$ by annotating each of the $500$ images, thereby obtaining a sample $y_i\sim Y$ associated with a sampled image $x_i\sim X$.}

    \part Show that finding the parameters $\theta$ that maximize the likelihood of your model over a sample of $\mathbf{d}$,
    \[
        p\left( (x_i,y_i) \mid \theta \right),
    \]
    is equivalent to finding the parameters $\theta$ that maximize the probability of the modelled conditional distribution over this sample,
    \[
        p\left( y_i \,\middle|\, x_i,\theta \right).
    \]

    \begin{solution}
        \begin{align*}
            \arg \max_{\theta} \left\{ p\left( (x_i, y_i) \mid \theta\right) \right\} &= \arg \max_{\theta} \left\{ p\left( y_i \mid x_i, \theta\right) p(x_i \mid \theta)\right\} \\
            &=\arg \max_{\theta} \left\{ p\left( y_i \mid x_i, \theta\right) p(x_i)\right\} \\
            & \qquad \, \text{(As the model is over $Y \mid X$, not $X$.)} \\
            &=\arg \max_{\theta} \left\{ p\left( y_i \mid x_i, \theta\right) \right\}
        \end{align*}
    \end{solution}

    \uplevel{After reviewing the literature, you choose a model for $p_\theta (y \mid x) = p(y \mid x, \theta)$.

    For the first task, you assume that a suitable model is to model $Y \mid X$ as a Gaussian distribution with homoscedastic variance. For the second task, a suitable model is a Bernoulli distribution. For the third task, a suitable model is a categorical distribution.}

    \part For each task, propose a suitable loss function to train your neural network. Propose also an output layer for your neural network in the form
    \[
        \mathbf{y} = \sigma_\text{out}\left(\mathbf{W}_y \mathbf{h} + \mathbf{b}_y\right).
    \]
    Let $\mathbf{h}$ denote the output of the layer immediately preceding the output layer, with dimension $d_o$ ($\mathbf{h} \in \mathbb{R}^{d_o}$). Specify the dimensions of $\mathbf{y}$, $\mathbf{W}_y$, and $\mathbf{b}_y$ and give an example of a suitable activation function $\sigma_\text{out}$ for each task.

    \textit{Hint:} For each task, which parameter of the distribution $p_\theta (y \mid x)$ is estimated by your neural network?

    \begin{solution}
        For the first task, the statistical model is:
        \begin{align*}
            Y \mid X &\sim \N\left(\mu (X), \sigma^2 \right) \\
            \rightarrow p(y \mid x) &\propto \exp\left(-\frac{1}{2}\frac{(y - \mu(x))^2}{\sigma^2} \right)
        \end{align*}
        The neural network estimates the mean of the Gaussian distribution and outputs $\hat{\mu}(X;\theta)$.

        For the second task:
        \begin{align*}
            Y \mid X &\sim \text{Bern}\left(q (X)\right) \\
            \rightarrow p(y \mid x) &\propto q(x)^{y}(1-q(x))^{(1-y)}
        \end{align*}
        The neural network estimates the probability that the image contains a dog and outputs $\hat{q}(X;\theta)$.

        For the third task, number the classes $0$ (dog), $1$ (cat) and $2$ (background). Each pixel $(u, v)$ has its own categorical distribution, the pixels being independent given the image:
        \begin{align*}
            P_{u,v} \mid X &\sim \text{Categ}\left(q_{0,uv}(X), q_{1,uv}(X), q_{2,uv}(X)\right) \\
            \rightarrow p(y \mid x) &= \prod_{u=0}^{31} \prod_{v=0}^{31} \prod_{c=0}^{2} q_{c,uv}(x)^{y_{c,uv}},
        \end{align*}
        with $\sum_c q_{c,uv}(x) = 1$, and where $y_{c,uv}$ is the one-hot encoding of the label: $1$ if pixel $(u, v)$ belongs to class $c$, $0$ otherwise. The neural network estimates the probabilities that each pixel belongs to a dog, a cat, or the background: it outputs $\hat{q}_{c,uv}(X;\theta)$ for every class $c$ and pixel $(u, v)$.

        From the statistical model, and assuming that the samples $(x_i,y_i)$ are independent and identically distributed, a loss function can be derived to maximize the likelihood of the model given the observed dataset $\mathbf{d}$:
        \begin{align*}
            \theta_{\text{MLE}} &= \arg \max_\theta p(\mathbf{d} \mid \theta) \\
            &= \arg \max_\theta p((x_1, y_1), \ldots, (x_N, y_N) \mid \theta) \\
            &= \arg \max _\theta \prod_{i=1}^{N} p(x_i, y_i \mid \theta ) \\
            &= \arg \max _\theta \prod_{i=1}^{N} p(y_i \mid x_i, \theta ) \\
            &= \arg \max _\theta \sum_{i=1}^{N} \log (p(y_i \mid x_i, \theta )) \\
            &= \arg \min _\theta \sum_{i=1}^{N} -\log (p(y_i \mid x_i, \theta )) \\
            &= \arg \min_\theta \mathcal{L}(\theta,\mathbf{d})
        \end{align*}

        For the first task with Gaussian likelihood, the loss is the mean squared error:
        \[
            \mathcal{L}(\theta,\mathbf{d}) = \frac{1}{N} \sum_{i=1}^{N} (y_i - \hat \mu (x_i ; \theta))^2
        \]

        For the second task with a Bernoulli likelihood, the loss is the binary cross-entropy:
        \[
            \mathcal{L}(\theta,\mathbf{d}) = -\frac{1}{N} \sum_{i=1}^{N} \left[ y_i \log(\hat q(x_i; \theta)) + (1-y_i)\log (1- \hat q(x_i;\theta)) \right]
        \]

        For the third task with categorical likelihood, the loss is the cross-entropy:
        \[
            \mathcal{L}(\theta,\mathbf{d}) = -\frac{1}{N} \sum_{i=1}^{N} \frac{1}{32 \times 32}\sum_{u=0}^{31} \sum_{v=0}^{31}\sum_{c=0}^2 y_{i, c, uv} \log(\hat q_{c,uv}(x_i; \theta)),
        \]

        where $y_{i,c,uv}$ is the one-hot encoding of the label of pixel $(u, v)$ in the $i$-th image.

        Regarding the output layer, the linear projection
        \[
            \mathbf{W}_y \mathbf{h} + \mathbf{b}_y
        \]
        must map the output $\mathbf{h}$ of the preceding layer of the neural network to the desired output dimension.

        For the first and the second task, $\mathbf{y} \in \mathbb{R}$, with
        \[
            \mathbf{W}_y \in \mathbb{R}^{1 \times d_o}, \quad \mathbf{b}_y \in \mathbb{R}.
        \]
        For the third task, the output holds one value per class and pixel, $3 \times 32 \times 32 = 3072$ values read as a tensor $\mathbf{y} \in \mathbb{R}^{3 \times 32 \times 32}$, with
        \[
            \mathbf{W}_y \in \mathbb{R}^{3072 \times d_o}, \quad \mathbf{b}_y \in \mathbb{R}^{3072}.
        \]
        The activation function $\sigma_\text{out}$ should map the projected output to the valid range of values for $\mathbf{y}$. For the first task, we can choose the identity:
        \[
            \sigma_\text{out}(z) = z.
        \]
        For the second task, we can choose a sigmoid activation to map the projected values between 0 and 1:
        \[
            \sigma_\text{out}(z) = \frac{1}{1+\exp(-z)}.
        \]
        For the third task, we can choose the softmax over the three classes of each pixel, to turn the logits $z_{c,uv}$ into probabilities:
        \[
            \hat q_{c,uv} = \frac{\exp(z_{c,uv})}{\sum_{c'=0}^2\exp(z_{c',uv})} \quad \text{for } c = 0, 1, 2 \text{ and every pixel } (u, v).
        \]
    \end{solution}

    \uplevel{You want to solve a new task: estimate the age of the dog or the cat in the image. To simplify the problem, you consider the age as an unrestricted real-valued quantity. After some reflection, it becomes clear that directly inferring the age from the image is not a good idea. If the type of animal (cat or dog) were first estimated, it would be much easier to estimate the animal's age from the image and this information.}

    \part Formulate the task in the context of supervised learning and explain all the elements appearing in $p_\theta(y \mid x)$. What mathematical quantity is estimated by the neural network?

    \begin{solution}
        $y \sim Y$ is the age of the animal in the picture ($Y \in \mathbb{R}$).

        $x \sim X$ is the picture ($X \in \mathbb{R}^{3 \times 32 \times 32}$).

        $\theta$ is the parameters of the model.

        The problem is a regression problem. The neural network should give an estimate of $\E\{ Y \mid X\}$.
    \end{solution}

    \part Draw a Bayesian network modelling the image, the type of animal and its age. Find a new expression for $p(y \mid x)$ that makes the effect of the type of the animal in the image explicit.

    \begin{solution}
        Let $A$ denote the random variable describing the type of animal present in the image (let's say $A=1$ if it is a dog). The type of the animal influences its age (cats and dogs do not have the same life span), and the animal and its age determine what the image shows. The image $X$ is observed.

        \begin{center}
            \begin{tikzpicture}[bn diagram]
                \node[random variable] (A) at (0, 0) {$A$};
                \node[random variable] (Y) at (2.4, 0) {$Y$};
                \node[observed variable] (X) at (1.2, -1.6) {$X$};
                \draw[dependency] (A) -- (Y);
                \draw[dependency] (A) -- (X);
                \draw[dependency] (Y) -- (X);
            \end{tikzpicture}
        \end{center}
        Any other ordering of the three variables, such as $X \to A$, $X \to Y$, $A \to Y$, gives a valid factorization of the same joint distribution, and the expression below holds whatever the network:

        \begin{align*}
            p(y \mid x) &= \sum _a p(y \mid A=a,x)p(A=a \mid x) \\
            &=p(y \mid A=1,x)p(A=1 \mid x) + p(y \mid A=0, x) (1-p(A=1 \mid x))
        \end{align*}
    \end{solution}

    \part Consider a large neural network designed to solve the task, divided into multiple submodules, each implemented as a separate neural network. Use the image classification model developed above as one of these submodules. Define the inputs and outputs of the additional submodules required to solve the task. Relate the resulting architecture to the mathematical expression derived above. Which submodule estimates each of the distributions appearing in this expression?

    \textit{Hint:} Derive an expression for $\E\{Y\mid X\}$ using the Bayesian network, explicitly accounting for the animal type $A$.

    \begin{solution}
        The neural network gives an estimate of the conditional expectation:
        \begin{align*}
            \E\{Y \mid X \} &= \E \{ \E \{Y \mid X, A \} \mid X \} \\
            &=\sum_a \E \{Y \mid X, A=a \} p(A=a \mid X) \\
            &=\E \{Y \mid X, A=1 \} p(A=1 \mid X) + \E \{Y \mid X, A=0 \} (1-p(A=1 \mid X))
        \end{align*}

        The expression can also be obtained directly by applying the expectation operator $\mathbb{E}_{Y\mid X}\{\cdot\}$ to both sides of the expression derived in (e).

        The expected age is therefore a weighted mean of the age inferred under the assumption that the image contains a dog and the age inferred under the assumption that it contains a cat. Each age is weighted by the corresponding probability.

        The first submodule is the image classification model developed previously. With parameters $\phi$, it takes an image $x$ as input and estimates the probability that the image contains a dog:
        \[
            \hat q (x;\phi) \approx p(A=1 \mid x) = 1 - p(A=0 \mid x)
        \]

        The second submodule, with parameters $\xi$, takes the image $x$ as input and estimates the expected age of the animal under the assumption that it is a dog:
        \[
            \hat \mu_d (x ; \xi) \approx \E\{Y\mid x,A=1\}
        \]

        A third submodule, with parameters $\psi$, takes the image $X$ as input and estimates the expected age under the assumption that it is a cat:
        \[
            \hat \mu_c (x ; \psi) \approx \E\{Y\mid x,A=0\}
        \]

        The final estimated age can then be reconstructed from the outputs of these submodules.
    \end{solution}
\end{parts}

\titledquestion{Neural network parameters}

Sandrine and Karl, two long-time data scientist friends, each decide to design a neural network to solve a challenging task: compressing a high-definition color image  in a HD 720p resolution ($1280\times720$ pixels).

To do so, both decide to use an encoder network and a decoder network. The first network, the \emph{encoder}, transforms the input image into a low-dimensional latent representation which constitutes the compressed representation. The second network, the \emph{decoder}, reconstructs the original image as faithfully as possible from this low-dimensional representation.

Sandrine decides to use a multilayer perceptron with ReLU nonlinear activations for her encoder. Each block $i$ of her encoder therefore consists of a linear projection followed by the nonlinear activation:
\[
    h_i = \sigma\left(W_i h_{i-1}+b_i\right).
\]

The HD color image is represented using three channels corresponding to the red, green, and blue components. Sandrine flattens the image into a single vector before feeding it to the network.

The dimensions of the different units throughout Sandrine's encoder are given below:

\begin{center}
    \includegraphics[width=0.4\linewidth]{figures/sandr_graph.png}
\end{center}

\begin{parts}
    \part How many trainable parameters does Sandrine's encoder contain? Is this network realistic to implement on a standard laptop? Assume that each parameter is encoded using $32$ bits.

    \begin{solution}
        The dimension of $W_i$ and $b_i$ for each layer are:
        \begin{enumerate}
            \item
            \begin{equation*}
                W_1 ^{921600 \times 2764800} \quad b_1^{921600}
            \end{equation*}

            \item
            \begin{equation*}
                W_2 ^{460800 \times 921600} \quad b_2^{460800}
            \end{equation*}

            \item
            \begin{equation*}
                W_3 ^{460800 \times 460800} \quad b_3^{460800}
            \end{equation*}

            \item
            \begin{equation*}
                W_4 ^{230400 \times 460800} \quad b_4^{230400}
            \end{equation*}
        \end{enumerate}
        The total number of parameters is:
        \begin{align*}
            N_{\mathrm{param}}
            &= 921600 \times 2764800 + 921600
               + 460800 \times 921600 + 460800 \\
            &\quad + 460800 \times 460800 + 460800
               + 230400 \times 460800 + 230400 \\
            &= \boxed{3\,291\,219\,993\,600}.
        \end{align*}

        Thus, using $32$ bits (4 bytes) per parameter, the memory required to store the parameters is
        \begin{align*}
            M
            &= 3\,291\,219\,993\,600 \times 4\ \text{bytes} = \frac{13\,164\,879\,974\,400}{2^{40}}\ \text{TiB} \\
            &\approx \boxed{11.97\ \text{TiB}}.
        \end{align*}
        Clearly, Sandrine's encoder is infeasible.
    \end{solution}

    \part What is the compression ratio achieved by Sandrine's encoder?

    \begin{solution}
        \begin{equation*}
            \eta = 1 - \frac{230400}{3\times 1280 \times 720} = 92 \%
        \end{equation*}
    \end{solution}

    \uplevel{Karl decides to use a convolutional neural network, also with ReLU nonlinear activations for his encoder.

    Each block of the network consists of a convolutional layer with $3\times3$ kernels, followed by a ReLU activation and, in every block but the last, a max-pooling layer with a pooling area of $2\times 2$. Before being passed through each block, the input feature maps of each block are padded with a one-pixel-wide border of zeros.

    The HD color image is still represented using three channels corresponding to the red, green, and blue components. The dimensions of the different units throughout Karl's encoder are given below:

    \begin{center}
        \includegraphics[width=0.6\linewidth]{figures/Karl_graph.png}
    \end{center}}

    \part How many parameters does Karl's encoder contain? Is this network realistic to implement on a standard laptop? Assume that each parameter is encoded using $32$ bits.

    \begin{solution}
        Let the input of a block be
        \[
            h \in \mathbb{R}^{a\times b\times c},
        \]
        where $a$ denotes the number of input channels, and $b$ and $c$ denote the height and width of the feature maps, respectively.

        Before the convolution, the input is padded with a border of zeros of width $1$. The padded input therefore has shape
        \[
            h_p \in \mathbb{R}^{a\times(b+2)\times(c+2)}.
        \]

        A convolutional kernel of spatial size $3\times3$ must span all $a$ input channels. Consequently, each kernel has shape
        \[
            a\times3\times3
        \]
        and therefore contains
        \[
            9a
        \]
        trainable weights. If the convolutional layer contains $k$ kernels, the total number of weights is
        \[
            \boxed{9ak}.
        \]
        As each kernel also has a bias, the total number of trainable parameters is
        \[
            \boxed{9ak+k=k(9a+1)}.
        \]

        Because the input is padded by one pixel on each side and the convolution uses a $3\times3$ kernel, the spatial dimensions are preserved:
        \[
            (b+2)-3+1=b,
            \qquad
            (c+2)-3+1=c.
        \]
        Therefore, the output of the convolutional layer has shape
        \[
            k\times b\times c.
        \]

        The ReLU does not modify the dimensions. Finally, the $2\times2$ max-pooling layer reduces each spatial dimension by a factor of $2$. Assuming $b$ and $c$ are even, the output of the block has shape
        \[
            \boxed{k\times\frac{b}{2}\times\frac{c}{2}},
        \]
        except for the last block, which has no pooling and keeps the shape $k \times b \times c$.
        The number of kernels and their shapes throughout the network are:
        \begin{enumerate}
            \item $4$ kernels of shape $3\times3\times3$,
            \item $8$ kernels of shape $4\times3\times3$,
            \item $32$ kernels of shape $8\times3\times3$,
            \item $16$ kernels of shape $32\times3\times3$.
        \end{enumerate}

        The total number of parameters is therefore
        \begin{align*}
            N_{\mathrm{param}}
            &=4(3\times3\times3+1)
              +8(4\times3\times3+1) \\
            &\quad
              +32(8\times3\times3+1)
              +16(32\times3\times3+1) \\
            &=4(28)+8(37)+32(73)+16(289) \\
            &=112+296+2336+4624 \\
            &=\boxed{7368}.
        \end{align*}
        Thus, using $32$ bits per parameter, the memory required to store the parameters is
        \begin{align*}
            M
            &= 7368\times4\ \text{bytes} = 29\,472\ \text{bytes} \\
            &= \frac{29\,472}{2^{10}}\ \text{KiB} \\
            &\approx \boxed{28.8\ \text{KiB}}.
        \end{align*}
        Karl's encoder clearly fits in a laptop.
    \end{solution}

    \part What is the compression ratio achieved by Karl's encoder?

    \begin{solution}
        \begin{equation*}
            \eta = 1 - \frac{16 \times 160 \times 90}{3\times 1280 \times 720} = 92 \%
        \end{equation*}
    \end{solution}
\end{parts}

\titledquestion{Learning to play Pacman \source{Exam, August 2020}}

You observe a Grandmaster agent playing Pacman. How can you use the moves you observe to train your own agent?

\begin{parts}
    \part Describe formally the data you would collect, the inference problem you would consider, and how you would solve it.

    \begin{solution}
        At each step $t$ of the games we watch, the Grandmaster perceives $e_t$ and plays $a_t \in \mathcal{A} = \{\text{North}, \text{South}, \text{East}, \text{West}, \text{Stop}\}$; Pacman's world is fully observable, so $e_t = s_t$. Each step gives an input-output pair $(\mathbf{x}_i, y_i)$, with features $\mathbf{x}_i = g(e_t) \in \mathbb{R}^d$ computed from the game state (positions of Pacman, the ghosts and the food, distances, walls around Pacman, \ldots) and the label $y_i = a_t$. Collecting $N$ steps gives a dataset $\mathbf{d} = \{(\mathbf{x}_1, y_1), \ldots, (\mathbf{x}_N, y_N)\}$.

        The inference problem is a classification problem with $|\mathcal{A}|$ classes: we look for a model $P_\theta(y \mid \mathbf{x})$ of the Grandmaster's choice given the features, whose parameters $\theta$ maximize the likelihood of $\mathbf{d}$,
        \[
            \theta^* = \arg\max_\theta \prod_{i=1}^N P_\theta(y_i \mid \mathbf{x}_i).
        \]
        Since there is no closed-form solution for a neural network, we minimize the negative log-likelihood by (stochastic) gradient descent. Our agent is then the simple reflex agent $\pi_\theta(e_t) = \arg\max_{a \in \mathcal{A}} P_{\theta^*}(a \mid g(e_t))$, whose condition-action rules are learned from the Grandmaster rather than written by hand.
    \end{solution}

    \part How would you design a neural network to control your agent? Define mathematically the neural network architecture, its inputs, its outputs, its parameters, as well as the loss you would use to train it.

    \begin{solution}
        A multilayer perceptron with input $\mathbf{x} = g(e_t) \in \mathbb{R}^d$, $K$ hidden layers and $C = |\mathcal{A}| = 5$ outputs:
        \begin{align*}
            \mathbf{h}_0 & = \mathbf{x}, \\
            \mathbf{h}_k & = \ReLU(\mathbf{W}_k \mathbf{h}_{k-1} + \mathbf{b}_k), \quad k = 1, \ldots, K, \\
            f(\mathbf{x}; \theta) & = \mathrm{Softmax}(\mathbf{W}_y \mathbf{h}_K + \mathbf{b}_y),
        \end{align*}
        with $\mathbf{W}_k \in \mathbb{R}^{q_k \times q_{k-1}}$, $\mathbf{b}_k \in \mathbb{R}^{q_k}$ ($q_0 = d$), $\mathbf{W}_y \in \mathbb{R}^{C \times q_K}$ and $\mathbf{b}_y \in \mathbb{R}^C$, so that $\theta = \{\mathbf{W}_1, \mathbf{b}_1, \ldots, \mathbf{W}_K, \mathbf{b}_K, \mathbf{W}_y, \mathbf{b}_y\}$. The output $f_a(\mathbf{x}; \theta)$ is the probability $P_\theta(a \mid \mathbf{x})$ that the Grandmaster plays $a$. Numbering the actions $0, \ldots, C-1$, the loss is the cross-entropy, the negative log-likelihood of $\mathbf{d}$:
        \[
            \mathcal{L}(\theta) = -\frac{1}{N} \sum_{i=1}^N \log f_{y_i}(\mathbf{x}_i; \theta).
        \]
        Illegal actions (into a wall) can be masked by setting their probability to zero before taking the $\arg\max$.
    \end{solution}

    \part Discuss the expected performance of the resulting agent when \begin{enumerate*}
        \item the Grandmaster agent is optimal, and
        \item the Grandmaster agent is suboptimal.
    \end{enumerate*}

    \begin{solution}
        \begin{enumerate*}
            \item If the Grandmaster is optimal, our agent is at best as good as the Grandmaster, and in practice worse. It only imitates the Grandmaster on the states the Grandmaster visits: once a small error leads it to a state that never appears in $\mathbf{d}$, its predictions there are unreliable and errors can compound over the rest of the game. More data, from varied situations, and features $g$ that generalize across mazes reduce this effect.
            \item If the Grandmaster is suboptimal, our agent learns its mistakes along with its good moves, since the loss only rewards agreeing with the Grandmaster. Nothing in the data tells which moves were good: doing better requires a signal about the outcome of the moves, such as the score, which is the setting of reinforcement learning.
        \end{enumerate*}
    \end{solution}
\end{parts}
