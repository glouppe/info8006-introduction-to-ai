\begin{recap}
    \topic{Kolmogorov's axioms}
    A \term{probability space} is a \term{sample space} $\Omega$, whose elements $\omega$ are the \term{sample points}, equipped with a probability function $P$ such that $0 \leq P(\omega) \leq 1$ and $\sum_{\omega \in \Omega} P(\omega) = 1$. An \term{event} is a set of sample points $A \subseteq \Omega$, of probability $P(A) = \sum_{\omega \in A} P(\omega)$. Hence $P(\Omega) = 1$, $P(A \cup B) = P(A) + P(B)$ for disjoint events, $P(\neg a) = 1 - P(a)$ and $P(a \lor b) = P(a) + P(b) - P(a \land b)$.

    \topic{Random variables}
    A \term{random variable} is a function $X : \Omega \to D_X$, with $P(X = x) = \sum_{\{\omega \,:\, X(\omega) = x\}} P(\omega)$. A number is written $P(x)$, short for $P(X = x)$; a whole \term{distribution} is written $\mathbf{P}(X)$, \ie{} the table of the values $P(X = x)$ for all $x \in D_X$.

    \topic{Conditional probability, sum and product rules}
    For $P(b) > 0$, the \term{conditional probability} of $a$ given $b$ is $P(a \mid b) = P(a, b) / P(b)$, hence the \term{product rule} $P(a, b) = P(a \mid b) P(b) = P(b \mid a) P(a)$. Summing a joint distribution over the values of a variable marginalizes it out (\term{sum rule}), $P(x) = \sum_y P(x, y)$, and with the product rule it gives the \term{law of total probability}, $P(x) = \sum_y P(x \mid y) P(y)$. Selecting the entries consistent with the evidence $\mathbf{e}$, marginalizing the hidden variables out and normalizing gives the posterior over a query variable $Q$, $\mathbf{P}(Q \mid \mathbf{e}) = \frac{1}{Z} \mathbf{P}(Q, \mathbf{e})$ with $Z = \sum_q P(q, \mathbf{e})$, the constant $1/Z$ being also written $\alpha$.

    \topic{Bayes' rule}
    From the two factorizations above,
    \[
        P(a \mid b) = \frac{P(b \mid a) P(a)}{P(b)},
    \]
    where $P(a)$ is the prior belief on $a$, $P(b)$ the probability of the evidence $b$, $P(a \mid b)$ the posterior belief on $a$ given $b$, and $P(b \mid a)$ the likelihood.

    \topic{Chain rule}
    Any joint distribution is an incremental product of conditional distributions, $P(x_1, \ldots, x_n) = \prod_{i=1}^n P(x_i \mid x_1, \ldots, x_{i-1})$.

    \topic{Independence}
    $A \perp B$ iff $P(a, b) = P(a) P(b)$, and $A \perp B \mid C$ iff $P(a, b \mid c) = P(a \mid c) P(b \mid c)$, for all values of the variables. Conditional independence factors a joint distribution into smaller tables, which is what makes probabilistic models scale.

\end{recap}

\titledquestion{Rational beliefs \aima{12}}

Let $A$ and $B$ be two \emph{events} over the probability space $\Omega$. An agent holds the beliefs $P(A) = 0.4$ and $P(B) = 0.3$. What ranges of probabilities would it be rational for the agent to hold for the events $A \cup B$ and $A \cap B$? What if the agent also believes that $P(A \mid B) = 0.5$?

\begin{solution}
    \begin{enumerate}
        \item The agent's beliefs should satisfy
        \begin{alignat*}{2}
            0.4 = \max \cbk{P(A), P(B)} & \leq P(A \cup B) && \leq \min \cbk{1, P(A) + P(B)} = 0.7 \\
            0 = \max \cbk{0, P(A) + P(B) - 1} & \leq P(A \cap B) && \leq \min \cbk{P(A), P(B)} = 0.3 .
        \end{alignat*}
        This can be illustrated with Venn diagrams.

        \item If the agent believes that $P(A \mid B) = 0.5$, it should also believe that
        \begin{equation*}
            P(A \cap B) = P(A \mid B) P(B) = 0.5 \times 0.3 = 0.15,
        \end{equation*}
        by the product rule. Then, the agent should rationally believe that
        \begin{equation*}
            P(A \cup B) = P(A) + P(B) - P(A \cap B) = 0.4 + 0.3 - 0.15 = 0.55.
        \end{equation*}
    \end{enumerate}
\end{solution}

\titledquestion{Full joint distribution \aima{12}}

A monitoring program supervises the bearings of a wind turbine. A worn bearing (\emph{fault}) tends to make the turbine vibrate (\emph{vibration}) and to trigger the alarm of a diagnostic probe (\emph{alarm}). The program holds the joint probability table below. Compute the following probabilities.

\begin{center}
    \begin{tabular}{lcccc}
        \toprule
        & \multicolumn{2}{c}{vibration} & \multicolumn{2}{c}{no vibration} \\
        \cmidrule(lr){2-3} \cmidrule(lr){4-5}
        & alarm & no alarm & alarm & no alarm \\
        \midrule
        fault & 0.108 & 0.012 & 0.072 & 0.008 \\
        no fault & 0.016 & 0.064 & 0.144 & 0.576 \\
        \bottomrule
    \end{tabular}
\end{center}

\begin{solution}
    In the following, we represent the events $\mathit{vibration}$, $\mathit{alarm}$ and $\mathit{fault}$ by the random variables $V, A, F: \Omega \to \cbk{0, 1}$, respectively. The table above is then the joint probability distribution $\mathbf{P}(V, A, F)$. When clear from context, we abbreviate $P(V = v, A = a, F = f)$ as $P(v, a, f)$.
\end{solution}

\begin{parts}
    \part $P(\mathit{vibration})$

    \begin{solution}
        \begin{equation*}
            P(V = 1) = \sum_{a} \sum_{f} P(V = 1, a, f) = 0.108 + 0.012 + 0.016 + 0.064 = 0.2
        \end{equation*}
    \end{solution}

    \part $P(\mathit{fault})$

    \begin{solution}
        \begin{equation*}
            P(F = 1) = \sum_{v} \sum_{a} P(v, a, F = 1) = 0.108 + 0.012 + 0.072 + 0.008 = 0.2
        \end{equation*}
    \end{solution}

    \part $P(\mathit{vibration} \mid \mathit{fault})$

    \begin{solution}
        \begin{align*}
            P(V = 1 \mid F = 1) = \frac{P(V = 1, F = 1)}{P(F = 1)} & = \frac{\sum_{a} P(V = 1, a, F = 1)}{P(F = 1)} \\
            & = \frac{0.108 + 0.012}{0.2} = 0.6
        \end{align*}
    \end{solution}

    \part $P(\mathit{fault} \mid \mathit{vibration} \cup \mathit{alarm})$

    \begin{solution}
        \begin{align*}
            P(F = 1 \mid V = 1 \cup A = 1) & = \frac{P(V = 1 \cup A = 1, F = 1)}{P(V = 1 \cup A = 1)} \\
            & = \frac{P(F = 1) - P(V = 0, A = 0, F = 1)}{1 - P(V = 0, A = 0)} \\
            & = \frac{0.2 - 0.008}{1 - 0.584} = \num{0.4615}
        \end{align*}
    \end{solution}
\end{parts}

\titledquestion{Fraud detection \aima{12}}

A fraud detection program screens card payments. It is \qty{99}{\percent} accurate, \ie{} it flags a fraudulent payment with probability \num{0.99} and a legitimate payment with probability \num{0.01}. Fraud is rare: it strikes only 1 payment in \num{10000}. The program has just flagged your payment. Why is it good news, for you, that fraud is rare? What is the probability that the payment is actually fraudulent?

\begin{solution}
    Let $f$ be the event that the payment is fraudulent and $a$ the event that it is flagged. Rarity is good news because the prior $P(f)$ is small: most flagged payments are legitimate payments flagged by mistake. Indeed,
    \begin{align*}
        P(f \mid a) & = \frac{P(a \mid f) P(f)}{P(a)} \\
        & = \frac{P(a \mid f) P(f)}{P(a \mid f) P(f) + P(a \mid \neg f) P(\neg f)} \\
        & = \frac{0.99 \times \num{e-4}}{0.99 \times \num{e-4} + 0.01 \times (1 - \num{e-4})} = \num{0.0098} .
    \end{align*}
\end{solution}

\titledquestion{Conditional independence \aima{12}}

For each of the following statements, either prove it is valid or give a counterexample.

\begin{parts}
    \part If $P(a, b, c) > 0$ and $P(a \mid b, c) = P(b \mid a, c)$, then $P(a \mid c) = P(b \mid c)$.

    \begin{solution}
        True. By the product rule, $P(a \mid b, c) P(b, c) = P(a, b, c) = P(b \mid a, c) P(a, c)$. Since $P(a, b, c) > 0$, $P(a \mid b, c) = P(b \mid a, c) > 0$, and dividing by it gives $P(b, c) = P(a, c)$ and thereby $P(b \mid c) = P(a \mid c)$.
    \end{solution}

    \part If $P(a \mid b, c) = P(a)$, then $P(b \mid c) = P(b)$.

    \begin{solution}
        False. The statement $P(a \mid b, c) = P(a)$ merely states that $a$ is independent of $b$ and $c$, it makes no claim regarding the dependence of $b$ and $c$. Counterexample: $a$ and $b$ record the results of two independent coin flips, and $c = b$.
    \end{solution}

    \part If $P(a \mid b) = P(a)$, then $P(a \mid b, c) = P(a \mid c)$.

    \begin{solution}
        False. While the statement $P(a \mid b) = P(a)$ implies that $a$ is independent of $b$, it does not imply that $a$ is conditionally independent of $b$ given $c$. Counterexample: $a$ and $b$ record the results of two independent coin flips, and $c = a \vee b$.
    \end{solution}
\end{parts}

\titledquestion{Bag of coins \aima{13}}

We have a bag of three biased coins $a$, $b$, and $c$ with probabilities of coming up heads of \qty{20}{\percent}, \qty{60}{\percent}, and \qty{80}{\percent}, respectively. One coin is drawn randomly from the bag (with equal probability of drawing each of the three coins), and then the coin is flipped three times to generate the outcomes $X_1$, $X_2$ and $X_3$ (heads or tails).

\begin{parts}
    \part Draw the Bayesian network corresponding to this setup and define the corresponding conditional probability tables (CPTs).

    \begin{solution}
        The process is described with the following random variables:
        \begin{itemize}
            \item $Y \in \cbk{a, b, c}$, the coin drawn from the bag.
            \item $X_i \in \cbk{h, t}$, the realization of the $i$-th coin toss.
        \end{itemize}
        The variable $Y$ is the parent of $X_i$, it should be put before in the network ordering.

        \begin{center}
            \begin{minipage}[c]{0.45\linewidth}
                \centering
                \begin{tikzpicture}[node distance = 2cm]
                    \node[state] (Y) {$Y$};
                    \node[state] (X1) [below left of=Y] {$X_1$};
                    \node[state] (X2) [below of=Y] {$X_2$};
                    \node[state] (X3) [below right of=Y] {$X_3$};

                    \draw[arrow] (Y) to (X1);
                    \draw[arrow] (Y) to (X2);
                    \draw[arrow] (Y) to (X3);
                \end{tikzpicture}
            \end{minipage}%
            \hfill
            \begin{minipage}[c]{0.45\linewidth}
                \centering
                \begin{tabular}{ccc}
                    \toprule
                    $Y$ & $P(Y)$ & $P(X_i = h \mid Y)$ \\
                    \midrule
                    $a$ & $\frac{1}{3}$ & $0.2$ \\
                    $b$ & $\frac{1}{3}$ & $0.6$ \\
                    $c$ & $\frac{1}{3}$ & $0.8$ \\
                    \bottomrule
                \end{tabular}
            \end{minipage}
        \end{center}
    \end{solution}

    \part Determine which coin ($a$, $b$ or $c$) is most likely to have been drawn from the bag if the observed tosses came out heads twice and tails once.

    \begin{solution}
        In order to find the most likely coin, we have to calculate the conditional probability distribution $P(Y \mid X_1 = h, X_2 = h, X_3 = t)$\footnote{It should be noted that the order of the tosses is irrelevant as they are performed independently.}. By Bayes' rule, we have
        \begin{equation*}
            P(Y \mid X_1 = h, X_2 = h, X_3 = t) = \alpha P(X_1 = h, X_2 = h, X_3 = t \mid Y) P(Y).
        \end{equation*}
        But we know that $X_1$, $X_2$ and $X_3$ are independent with respect to $Y$, meaning that
        \begin{equation*}
            P(X_1, X_2, X_3 \mid Y) = P(X_1 \mid Y) P(X_2 \mid Y) P(X_3 \mid Y).
        \end{equation*}
        Then, we have
        \begin{equation*}
            P(Y \mid X_1 = h, X_2 = h, X_3 = t) = \alpha \times \frac{1}{3} \times \begin{cases}
                0.2 \times 0.2 \times 0.8 = \num{0.032} & \text{if } Y = a \\
                0.6 \times 0.6 \times 0.4 = \num{0.144} & \text{if } Y = b \\
                0.8 \times 0.8 \times 0.2 = \num{0.128} & \text{if } Y = c \\
            \end{cases}
        \end{equation*}
        meaning that $b$ is the most likely coin to have been drawn. Interestingly, we don't need to normalize the distribution as we only care about the maximum.
    \end{solution}
\end{parts}

\titledquestion{Evolutionary model merging \aima{13}}

Evolutionary model merging builds a new language model $M$ from two parent models $A$ and $B$: each component of $M$ is copied from one of the parents. Consider one component, and let $G_x$ be the language it was mostly trained on in model $x$, with possible values $e$ (English) or $f$ (French). A common \emph{hypothesis} is that the default language of a model follows from this component by a simple mechanism: the language $H_x$ in which model $x$ answers by default is the same as $G_x$ with some probability $s$.

Furthermore, $M$ copies the component from either parent with equal probability, and a random perturbation flips it with a small nonzero probability $m$.

\begin{parts}
    \part Which of the following networks claim that $P(G_M \mid G_A, G_B) = P(G_M)$? Shaded variables are observed.

    \begin{center}
        \begin{tikzpicture}[bn diagram]
            \node[random variable] (GA1) at (0, 0) {$G_A$};
            \node[random variable] (GB1) at (3.2, 0) {$G_B$};
            \node[observed variable] (HA1) at (0, -1.4) {$H_A$};
            \node[observed variable] (HB1) at (3.2, -1.4) {$H_B$};
            \node[random variable] (GM1) at (1.6, -2) {$G_M$};
            \node[observed variable] (HM1) at (1.6, -3.4) {$H_M$};
            \draw[dependency] (GA1) -- (HA1);
            \draw[dependency] (GB1) -- (HB1);
            \draw[dependency] (GA1) -- (GM1);
            \draw[dependency] (GB1) -- (GM1);
            \draw[dependency] (GM1) -- (HM1);
            \node[diagram caption] at (1.6, -4.3) {(1)};
            \node[random variable] (GA2) at (5, 0) {$G_A$};
            \node[random variable] (GB2) at (8.2, 0) {$G_B$};
            \node[observed variable] (HA2) at (5, -1.4) {$H_A$};
            \node[observed variable] (HB2) at (8.2, -1.4) {$H_B$};
            \node[random variable] (GM2) at (6.6, -2) {$G_M$};
            \node[observed variable] (HM2) at (6.6, -3.4) {$H_M$};
            \draw[dependency] (GA2) -- (HA2);
            \draw[dependency] (GB2) -- (HB2);
            \draw[dependency] (GA2) -- (GM2);
            \draw[dependency] (GB2) -- (GM2);
            \draw[dependency] (HA2) -- (HM2);
            \draw[dependency] (GM2) -- (HM2);
            \draw[dependency] (HB2) -- (HM2);
            \node[diagram caption] at (6.6, -4.3) {(2)};
            \node[random variable] (GA3) at (10, 0) {$G_A$};
            \node[random variable] (GB3) at (13.2, 0) {$G_B$};
            \node[observed variable] (HA3) at (10, -1.4) {$H_A$};
            \node[observed variable] (HB3) at (13.2, -1.4) {$H_B$};
            \node[random variable] (GM3) at (11.6, -2) {$G_M$};
            \node[observed variable] (HM3) at (11.6, -3.4) {$H_M$};
            \draw[dependency] (GA3) -- (HA3);
            \draw[dependency] (GB3) -- (HB3);
            \draw[dependency] (HA3) -- (HM3);
            \draw[dependency] (GM3) -- (HM3);
            \draw[dependency] (HB3) -- (HM3);
            \node[diagram caption] at (11.6, -4.3) {(3)};
        \end{tikzpicture}
    \end{center}

    \begin{solution}
        The third one, as the node $G_M$ does not have any parents.
    \end{solution}

    \part Which of the networks make independence claims that are consistent with the hypothesis about the default language?

    \begin{solution}
        The first and second ones. The second one is also valid as it makes \emph{fewer} assumptions about the dependencies between the variables, \ie{} it is more general than the first one.
    \end{solution}

    \part Write down the CPT for the $G_M$ node in the first network, in terms of $s$ and $m$.

    \begin{solution}
        \begin{center}
            \begin{tabular}{ccc}
                \toprule
                $G_A$ & $G_B$ & $P(G_M = f \mid G_A, G_B)$ \\
                \midrule
                $f$ & $f$ & $1 - m$ \\
                $f$ & $e$ & $0.5$ \\
                $e$ & $f$ & $0.5$ \\
                $e$ & $e$ & $m$ \\
                \bottomrule
            \end{tabular}
        \end{center}
    \end{solution}

    \part Suppose that $P(G_A = f) = P(G_B = f) = q$. In the first network, derive an expression for $P(G_M = f)$ in terms of $s$, $m$ and $q$.

    \begin{solution}
        \begin{align*}
            P(G_M = f) & = \sum_{g_A} \sum_{g_B} P(G_M = f, g_A, g_B) \\
            & = \sum_{g_A} \sum_{g_B} P(G_M = f \mid g_A, g_B) P(g_A) P(g_B) \\
            & = (1 - m) q^2 + 0.5 (1 - q) q + 0.5 q (1 - q) + m (1 - q)^2 \\
            & = q + m - 2qm
        \end{align*}
    \end{solution}

    \part Over many generations of merges, we expect the distribution of the component to be the same in the parents and in the merged models. Use this to calculate the value of $q$. Only about \qty{10}{\percent} of the merged models answer in French by default: explain why the hypothesis described at the beginning of this question must be wrong.

    \begin{solution}
        At equilibrium, $P(G_M = f) = P(G_A = f) = q$. Therefore, $q = q + m - 2 q m$, meaning that $q = 0.5$. Answering in French is much rarer than answering in English, which invalidates the hypothesis.
    \end{solution}
\end{parts}

\begin{quiz}
    \item The Bayes' rule states that \ldots
    \begin{enumerate}[label=$\bigcirc$]
        \item $P(x \mid y) = \dfrac{P(y \mid x) P(y)}{P(x)}$.
        \item \correct{$P(x \mid y) = \dfrac{P(y \mid x) P(x)}{P(y)}$.}
        \item $P(y \mid x) = \dfrac{P(x \mid y) P(x)}{P(y)}$.
        \item $P(x) = \dfrac{P(x \mid y) P(y)}{P(x)}$.
    \end{enumerate}

    \item Which of the following is true?
    \begin{enumerate}[label=$\bigcirc$]
        \item $P(x, y \mid z) = \dfrac{P(x \mid y, z)}{P(y, z \mid x)}$.
        \item $P(x, y, z) = P(x) P(y) P(z)$.
        \item \correct{$P(x, y \mid z) = P(y \mid x, z) P(x \mid z)$.}
        \item $P(x \mid y, z) = \dfrac{P(x, y, z)}{P(y) P(z)}$.
    \end{enumerate}

    \item Naive Bayes model assumes \ldots
    \begin{enumerate}[label=$\bigcirc$]
        \item \correct{conditional independence between the effects, given the cause.}
        \item independence between the effects.
        \item independence between the effects and the cause.
        \item conditional independence between each effect and the cause, given the other effects.
    \end{enumerate}

    \item The Bayes' rule can be read as ``The posterior is equal to \ldots
    \begin{enumerate}[label=$\bigcirc$]
        \item the product of the likelihood and the prior''.
        \item the ratio between the posterior distribution and the evidence, times the prior''.
        \item the likelihood to prior ratio multiplied by the evidence''.
        \item \correct{the likelihood to evidence ratio multiplied by the prior''.}
    \end{enumerate}
\end{quiz}
