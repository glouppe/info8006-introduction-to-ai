\begin{recap}
    \topic{Definitions}
    A probability is a degree of belief in a proposition.
    \begin{center}
        \begin{tabular}{ll}
            \toprule
            Sample space & $\Omega$, of possible worlds $\omega$ \\
            Axioms & $0 \leq P(\omega) \leq 1$, \quad $\sum_{\omega \in \Omega} P(\omega) = 1$ \\
            Event & $A \subseteq \Omega$, \quad $P(A) = \sum_{\omega \in A} P(\omega)$ \\
            Random variable & $X : \Omega \to D_X$, with $X = x$ the event $\{\omega \in \Omega : X(\omega) = x\}$ \\
            Distribution & $\mathbf{P}(X)$, the numbers $P(x) = P(X = x)$ for $x \in D_X$ \\
            Joint distribution & $\mathbf{P}(X_1, \ldots, X_n)$, the numbers $P(x_1, \ldots, x_n)$ \\
            Conditional probability & $P(a \mid b) = P(a, b) / P(b)$, for $P(b) > 0$ \\
            Conditional distribution & $\mathbf{P}(X \mid Y)$, one $\mathbf{P}(X \mid y)$ for each $y \in D_Y$ \\
            \bottomrule
        \end{tabular}
    \end{center}
    Random variables are written in upper case and their values in lower case; a boolean variable $A$ takes the values $a$ and $\neg a$. $P(x)$ is a number, $\mathbf{P}(X)$ a table.

    \topic{Rules}
    \begin{center}
        \begin{tabular}{ll}
            \toprule
            Complement & $P(\neg a) = 1 - P(a)$ \\
            Union & $P(a \lor b) = P(a) + P(b) - P(a \land b)$ \\
            Product rule & $P(a, b) = P(a \mid b) P(b) = P(b \mid a) P(a)$ \\
            Sum rule, total probability & $P(x) = \sum_y P(x, y) = \sum_y P(x \mid y) P(y)$ \\
            Chain rule & $P(x_{1:n}) = \prod_i P(x_i \mid x_{1:i-1})$ \\
            Bayes' rule & $P(a \mid b) = P(b \mid a) P(a) / P(b)$ \\
            Independence, $X \perp Y$ & $P(x, y) = P(x) P(y)$ \\
            Conditional independence, $X \perp Y \mid Z$ & $P(x, y \mid z) = P(x \mid z) P(y \mid z)$ \\
            Naive Bayes & $P(c, e_{1:n}) = P(c) \prod_i P(e_i \mid c)$ \\
            \bottomrule
        \end{tabular}
    \end{center}
    The rules hold for all values, and between distributions. Naive Bayes assumes the effects $e_{1:n}$ conditionally independent given the cause $c$. The joint distribution answers every query: select the entries consistent with the evidence, sum out the other variables, normalize.

    \topic{Bayes' rule}
    A model is given in the \term{causal} direction, a \term{prior} $\mathbf{P}(A)$ and a \term{likelihood} $\mathbf{P}(B \mid A)$. The \term{diagnostic} question is answered by the \term{posterior} $\mathbf{P}(A \mid b) = \frac{1}{Z} \mathbf{P}(b \mid A) \mathbf{P}(A)$, normalized by the probability of the \term{evidence}, ${Z = P(b) = \sum_a P(b \mid a) P(a)}$.
\end{recap}

\titledquestion{Rational beliefs \aima{12}}

Let $A, B \subseteq \Omega$ be two \emph{events}. An agent holds the beliefs $P(A) = 0.4$ and $P(B) = 0.3$. What ranges of probabilities would it be rational for the agent to hold for the events $A \cup B$ and $A \cap B$? What if the agent also believes that $P(A \mid B) = 0.5$?

\begin{solution}
    \begin{enumerate}
        \item Since $P(E) = \sum_{\omega \in E} P(\omega)$ sums nonnegative terms, $E \subseteq F$ implies $P(E) \leq P(F)$. With $A, B \subseteq A \cup B$ and $A \cap B \subseteq A, B$, and the union rule $P(A \cup B) = P(A) + P(B) - P(A \cap B)$, the agent's beliefs should satisfy
        \begin{alignat*}{2}
            0.4 = \max \cbk{P(A), P(B)} & \leq P(A \cup B) && \leq \min \cbk{1, P(A) + P(B)} = 0.7 \\
            0 = \max \cbk{0, P(A) + P(B) - 1} & \leq P(A \cap B) && \leq \min \cbk{P(A), P(B)} = 0.3 .
        \end{alignat*}
        The two beliefs are not independent choices: by the union rule, $P(A \cup B) + P(A \cap B) = 0.7$. The extremes are $B \subseteq A$, with $P(A \cup B) = 0.4$ and $P(A \cap B) = 0.3$, and disjoint events, with $0.7$ and $0$.

        \item If the agent believes that $P(A \mid B) = 0.5$, it should also believe that
        \begin{equation*}
            P(A \cap B) = P(A \mid B) P(B) = 0.5 \times 0.3 = 0.15,
        \end{equation*}
        by the product rule. Then, the agent should rationally believe that
        \begin{equation*}
            P(A \cup B) = P(A) + P(B) - P(A \cap B) = 0.4 + 0.3 - 0.15 = 0.55.
        \end{equation*}
    \end{enumerate}
\end{solution}

\titledquestion{Full joint distribution \aima{12}}

A monitoring program supervises the bearings of a wind turbine. A worn bearing (\emph{fault}) tends to make the turbine vibrate (\emph{vibration}) and to trigger the alarm of a diagnostic probe (\emph{alarm}). The program holds the joint distribution $\mathbf{P}(V, A, F)$ below, over the boolean random variables $V$ (vibration), $A$ (alarm) and $F$ (fault). Compute the following probabilities.

\begin{center}
    \begin{tabular}{lcccc}
        \toprule
        & \multicolumn{2}{c}{$v$} & \multicolumn{2}{c}{$\neg v$} \\
        \cmidrule(lr){2-3} \cmidrule(lr){4-5}
        & $a$ & $\neg a$ & $a$ & $\neg a$ \\
        \midrule
        $f$ & 0.108 & 0.012 & 0.072 & 0.008 \\
        $\neg f$ & 0.016 & 0.064 & 0.144 & 0.576 \\
        \bottomrule
    \end{tabular}
\end{center}

\begin{parts}
    \part $P(v)$

    \begin{solution}
        \begin{equation*}
            P(v) = \sum_{a'} \sum_{f'} P(v, a', f') = 0.108 + 0.012 + 0.016 + 0.064 = 0.2
        \end{equation*}
    \end{solution}

    \part $P(f)$

    \begin{solution}
        \begin{equation*}
            P(f) = \sum_{v'} \sum_{a'} P(v', a', f) = 0.108 + 0.012 + 0.072 + 0.008 = 0.2
        \end{equation*}
    \end{solution}

    \part $P(v \mid f)$

    \begin{solution}
        \begin{equation*}
            P(v \mid f) = \frac{P(v, f)}{P(f)} = \frac{\sum_{a'} P(v, a', f)}{P(f)} = \frac{0.108 + 0.012}{0.2} = 0.6
        \end{equation*}
    \end{solution}

    \part $P(f \mid v \lor a)$

    \begin{solution}
        \begin{align*}
            P(f \mid v \lor a) & = \frac{P(f, v \lor a)}{P(v \lor a)} = \frac{P(f) - P(\neg v, \neg a, f)}{1 - P(\neg v, \neg a)} \\
            & = \frac{0.2 - 0.008}{1 - 0.584} = \num{0.4615}
        \end{align*}
    \end{solution}

    \part Are $V$ and $A$ independent? Are they conditionally independent given $F$?

    \begin{solution}
        With $P(a) = 0.108 + 0.016 + 0.072 + 0.144 = 0.34$, $P(v, a) = 0.108 + 0.016 = 0.124 \neq P(v) P(a) = 0.2 \times 0.34 = 0.068$: $V$ and $A$ are not independent. Given $F$, for boolean variables it suffices to check one entry per value of $F$, the others following by complement:
        \begin{align*}
            P(v, a \mid f) & = \frac{0.108}{0.2} = 0.54 = P(v \mid f) P(a \mid f) = 0.6 \times 0.9, \\
            P(v, a \mid \neg f) & = \frac{0.016}{0.8} = 0.02 = P(v \mid \neg f) P(a \mid \neg f) = 0.1 \times 0.2,
        \end{align*}
        so $V \perp A \mid F$. The table is a naive Bayes model, with the fault as the cause and the vibration and the alarm as its effects: $\mathbf{P}(V, A, F) = \mathbf{P}(F) \mathbf{P}(V \mid F) \mathbf{P}(A \mid F)$ needs $1 + 2 + 2 = 5$ numbers instead of 7. Vibration and alarm are dependent only through their common cause.
    \end{solution}
\end{parts}

\titledquestion{Fraud detection \aima{12}}

A fraud detection program screens card payments. It is \qty{99}{\percent} accurate, \ie{} it flags a fraudulent payment with probability \num{0.99} and a legitimate payment with probability \num{0.01}. Fraud is rare: it strikes only 1 payment in \num{10000}. The program has just flagged your payment. Why is it good news, for you, that fraud is rare? What is the probability that the payment is actually fraudulent?

\begin{solution}
    Let $f$ be the event that the payment is fraudulent and $a$ the event that it is flagged. Rarity is good news because the prior $P(f)$ is small: most flagged payments are legitimate payments flagged by mistake. Indeed,
    \begin{align*}
        P(f \mid a) & = \frac{P(a \mid f) P(f)}{P(a)} \\
        & = \frac{P(a \mid f) P(f)}{P(a \mid f) P(f) + P(a \mid \neg f) P(\neg f)} \\
        & = \frac{0.99 \times \num{e-4}}{0.99 \times \num{e-4} + 0.01 \times (1 - \num{e-4})} = \num{0.0098} .
    \end{align*}
\end{solution}

\titledquestion{Conditional independence \aima{12}}

For each of the following statements, either prove it is valid or give a counterexample.

\begin{parts}
    \part If $P(a, b, c) > 0$ and $P(a \mid b, c) = P(b \mid a, c)$, then $P(a \mid c) = P(b \mid c)$.

    \begin{solution}
        True. By the product rule, $P(a \mid b, c) P(b, c) = P(a, b, c) = P(b \mid a, c) P(a, c)$. Since $P(a, b, c) > 0$, $P(a \mid b, c) = P(b \mid a, c) > 0$, and dividing by it gives $P(b, c) = P(a, c)$ and thereby $P(b \mid c) = P(a \mid c)$.
    \end{solution}

    \part If $P(a \mid b, c) = P(a)$, then $P(b \mid c) = P(b)$.

    \begin{solution}
        False. The statement $P(a \mid b, c) = P(a)$ merely states that $a$ is independent of $b$ and $c$, it makes no claim regarding the dependence of $b$ and $c$. Counterexample: $a$ and $b$ record the results of two independent coin flips, and $c = b$.
    \end{solution}

    \part If $P(a \mid b) = P(a)$, then $P(a \mid b, c) = P(a \mid c)$.

    \begin{solution}
        False. While the statement $P(a \mid b) = P(a)$ implies that $a$ is independent of $b$, it does not imply that $a$ is conditionally independent of $b$ given $c$. Counterexample: $a$ and $b$ record the results of two independent coin flips, and $c = a \vee b$.
    \end{solution}
\end{parts}

\titledquestion{Driving styles \aima{13}}

A self-driving car shares the highway with another car, whose driving style it does not know. It models three styles, cautious ($a$), normal ($b$) and aggressive ($c$), equally likely a priori, under which the other car changes lanes during a given minute with probabilities \num{0.2}, \num{0.6} and \num{0.8}, respectively. The self-driving car observes the other car for three minutes, recording whether it changes lanes ($l$) or keeps its lane ($k$) during each minute, as $X_1$, $X_2$ and $X_3$. Let $Y$ be the style of the other car.

\begin{parts}
    \part Write the joint distribution $\mathbf{P}(Y, X_1, X_2, X_3)$ of this model as a product of smaller tables, and give these tables.

    \begin{solution}
        The process is described with the following random variables:
        \begin{itemize}
            \item $Y \in \cbk{a, b, c}$, the driving style of the other car;
            \item $X_i \in \cbk{l, k}$, its behaviour during the $i$-th minute.
        \end{itemize}
        The observations are conditionally independent given the style, so the model is a naive Bayes model, with the style as the cause and the observations as its effects:
        \begin{equation*}
            \mathbf{P}(Y, X_1, X_2, X_3) = \mathbf{P}(Y) \mathbf{P}(X_1 \mid Y) \mathbf{P}(X_2 \mid Y) \mathbf{P}(X_3 \mid Y),
        \end{equation*}
        with the same table $\mathbf{P}(X_i \mid Y)$ for the three minutes:
        \begin{center}
            \begin{tabular}{ccc}
                \toprule
                $y$ & $P(y)$ & $P(X_i = l \mid y)$ \\
                \midrule
                $a$ & $1/3$ & $0.2$ \\
                $b$ & $1/3$ & $0.6$ \\
                $c$ & $1/3$ & $0.8$ \\
                \bottomrule
            \end{tabular}
        \end{center}
        The two tables hold $2 + 3 = 5$ free numbers, against $3 \times 2^3 - 1 = 23$ for the full joint distribution.
    \end{solution}

    \part Determine the most likely driving style if the other car changed lanes during the first two minutes and kept its lane during the third.

    \begin{solution}
        We compute the posterior $\mathbf{P}(Y \mid X_1 = l, X_2 = l, X_3 = k)$. By Bayes' rule,
        \begin{equation*}
            \mathbf{P}(Y \mid X_1 = l, X_2 = l, X_3 = k) = \frac{1}{Z} \mathbf{P}(X_1 = l, X_2 = l, X_3 = k \mid Y) \mathbf{P}(Y).
        \end{equation*}
        With the factorization of (a),
        \begin{equation*}
            P(y \mid X_1 = l, X_2 = l, X_3 = k) = \frac{1}{Z} \times \frac{1}{3} \times \begin{cases}
                0.2 \times 0.2 \times 0.8 = \num{0.032} & \text{if } y = a \\
                0.6 \times 0.6 \times 0.4 = \num{0.144} & \text{if } y = b \\
                0.8 \times 0.8 \times 0.2 = \num{0.128} & \text{if } y = c,
            \end{cases}
        \end{equation*}
        with $Z = \frac{1}{3} (0.032 + 0.144 + 0.128)$. The posterior is $\langle 0.105, 0.474, 0.421 \rangle$: the normal style is the most likely, though the aggressive one is nearly as probable. Finding the most likely style only needs the unnormalized values. The order of the observations does not matter: any two lane changes and one lane keep have the same probability given each style.
    \end{solution}
\end{parts}

\begin{quiz}
    \item The Bayes' rule states that \ldots
    \begin{enumerate}[label=$\bigcirc$]
        \item $P(x \mid y) = \dfrac{P(y \mid x) P(y)}{P(x)}$.
        \item \correct{$P(x \mid y) = \dfrac{P(y \mid x) P(x)}{P(y)}$.}
        \item $P(y \mid x) = \dfrac{P(x \mid y) P(x)}{P(y)}$.
        \item $P(x) = \dfrac{P(x \mid y) P(y)}{P(x)}$.
    \end{enumerate}

    \item Which of the following is true?
    \begin{enumerate}[label=$\bigcirc$]
        \item $P(x, y \mid z) = \dfrac{P(x \mid y, z)}{P(y, z \mid x)}$.
        \item $P(x, y, z) = P(x) P(y) P(z)$.
        \item \correct{$P(x, y \mid z) = P(y \mid x, z) P(x \mid z)$.}
        \item $P(x \mid y, z) = \dfrac{P(x, y, z)}{P(y) P(z)}$.
    \end{enumerate}

    \item The naive Bayes model assumes \ldots
    \begin{enumerate}[label=$\bigcirc$]
        \item \correct{conditional independence between the effects, given the cause.}
        \item independence between the effects.
        \item independence between the effects and the cause.
        \item conditional independence between each effect and the cause, given the other effects.
    \end{enumerate}

    \item The Bayes' rule can be read as ``The posterior is equal to \ldots
    \begin{enumerate}[label=$\bigcirc$]
        \item the product of the likelihood and the prior''.
        \item the ratio between the posterior distribution and the evidence, times the prior''.
        \item the likelihood to prior ratio multiplied by the evidence''.
        \item \correct{the likelihood to evidence ratio multiplied by the prior''.}
    \end{enumerate}
\end{quiz}
